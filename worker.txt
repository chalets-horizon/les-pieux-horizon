async function anthropicCall(env, body, state) {
  const models = state.model ? [state.model] : MODELS;
  let lastError = null;
  for (const model of models) {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' },
      body: JSON.stringify({ model, ...body }),
    });
    const data = await res.json().catch(() => ({}));
    if (res.ok && Array.isArray(data.content)) { state.model = model; return { ok: true, data }; }
    lastError = { model, status: res.status, type: data.error && data.error.type, message: data.error && data.error.message };
    console.error('Erreur API Anthropic', JSON.stringify(lastError));
    // 404 = modèle inconnu/retiré → on tente le suivant. Autre erreur (401 clé, 400 crédit, 429, 529…) : inutile d'insister.
    if (res.status !== 404) break;
  }
  return { ok: false, error: lastError };
}

// Boucle : réponse → (outil → résultat → réponse)* ; renvoie le texte final et la réservation éventuelle
async function runClaude(env, system, messages, convId) {
  const msgs = messages.map(m => ({ role: m.role === 'assistant' ? 'assistant' : 'user', content: String(m.content || '') }));
  const state = { model: null };
  let booking = null;
  let text = '';
  for (let round = 0; round < 4; round++) {
    const r = await anthropicCall(env, { max_tokens: 700, system, messages: msgs, tools: CHAT_TOOLS }, state);
    if (!r.ok) return { ok: false, error: r.error };
    const content = r.data.content;
    text = content.filter(b => b.type === 'text').map(b => b.text).join('\n').trim();
    const uses = content.filter(b => b.type === 'tool_use');
    if (!uses.length || r.data.stop_reason !== 'tool_use') break;
    msgs.push({ role: 'assistant', content });
    const results = [];
    for (const u of uses) {
      let out;
      try {
        if (u.name === 'chercher_client') out = await clientLookup(env, u.input || {}, convId);
        else if (u.name === 'creneaux_visite') out = await agendaSlots(env, u.input || {}, convId, 'chatbot');
        else if (u.name === 'reserver_visite') { out = await agendaBook(env, u.input || {}, convId, 'chatbot'); if (out.ok) booking = { ...u.input, libelle: out.libelle, lien: out.lien }; }
        else out = { ok: false, raison: 'outil inconnu' };
      } catch (e) { out = { ok: false, raison: String(e && e.message ? e.message : e) }; console.error('Outil', u.name, String(e && e.message ? e.message : e)); }
      results.push({ type: 'tool_result', tool_use_id: u.id, content: JSON.stringify(out) });
    }
    msgs.push({ role: 'user', content: results });
  }
  if (!text) text = booking ? 'C\'est noté, votre visite est réservée le ' + booking.libelle + '.' : FALLBACK_REPLY;
  return { ok: true, text, booking };
}

function parisParts(d) {
  const f = new Intl.DateTimeFormat('en-GB', { timeZone: 'Europe/Paris', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', weekday: 'short', hour12: false });
  const p = {}; for (const x of f.formatToParts(d)) p[x.type] = x.value;
  return { y: +p.year, m: +p.month, d: +p.day, h: (+p.hour) % 24, mi: +p.minute, wd: ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].indexOf(p.weekday) };
}
function parisOffsetMin(d) { const p = parisParts(d); return Math.round((Date.UTC(p.y, p.m - 1, p.d, p.h, p.mi) - Math.floor(d.getTime() / 60000) * 60000) / 60000); }
// Heure locale de Paris → Date (instant)
function parisDate(y, m, d, hh, mm) {
  const naive = Date.UTC(y, m - 1, d, hh, mm);
  const off = parisOffsetMin(new Date(naive - 60 * 60000));
  return new Date(naive - off * 60000);
}
function addDays(y, m, d, n) { const t = new Date(Date.UTC(y, m - 1, d + n)); return { y: t.getUTCFullYear(), m: t.getUTCMonth() + 1, d: t.getUTCDate() }; }

function easter(y) { const a = y % 19, b = Math.floor(y / 100), c = y % 100, d = Math.floor(b / 4), e = b % 4, f = Math.floor((b + 8) / 25), g = Math.floor((b - f + 1) / 3), h = (19 * a + b - d - g + 15) % 30, i = Math.floor(c / 4), k = c % 4, l = (32 + 2 * e + 2 * i - h - k) % 7, m = Math.floor((a + 11 * h + 22 * l) / 451), mo = Math.floor((h + l - 7 * m + 114) / 31), da = ((h + l - 7 * m + 114) % 31) + 1; return { m: mo, d: da }; }
function isHoliday(y, m, d) {
  const fixed = ['1-1', '5-1', '5-8', '7-14', '8-15', '11-1', '11-11', '12-25'];
  if (fixed.includes(m + '-' + d)) return true;
  const e = easter(y);
  for (const n of [1, 39, 50]) { const t = addDays(y, e.m, e.d, n); if (t.m === m && t.d === d) return true; }
  return false;
}

function haversineKm(a, b) {
  const R = 6371, dLat = (b.lat - a.lat) * Math.PI / 180, dLon = (b.lon - a.lon) * Math.PI / 180;
  const s = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * Math.PI / 180) * Math.cos(b.lat * Math.PI / 180) * Math.sin(dLon / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(s));
}
// Estimation sans itinéraire : distance à vol d'oiseau corrigée (utilisée si l'API Routes n'est pas configurée)
function driveMinutes(geo) { return Math.round(10 + haversineKm(HOME, geo) * 1.25 / 80 * 60); }

      });

      const route = j.routes && j.routes[0];
      if (r.ok && route && route.duration) out = { min: Math.round(parseInt(route.duration, 10) / 60), km: Math.round((route.distanceMeters || 0) / 1000), source: 'maps' };

function sectorOf(geo, cp) {
  const km = haversineKm(HOME, geo);
  if (km < 30) return 'proche';
  const y = Math.sin((geo.lon - HOME.lon) * Math.PI / 180) * Math.cos(geo.lat * Math.PI / 180);
  const x = Math.cos(HOME.lat * Math.PI / 180) * Math.sin(geo.lat * Math.PI / 180) - Math.sin(HOME.lat * Math.PI / 180) * Math.cos(geo.lat * Math.PI / 180) * Math.cos((geo.lon - HOME.lon) * Math.PI / 180);
  const brg = (Math.atan2(y, x) * 180 / Math.PI + 360) % 360;
  const dir = ['N', 'NE', 'E', 'SE', 'S', 'SO', 'O', 'NO'][Math.round(brg / 45) % 8];
  return String(cp || '').slice(0, 2) + '-' + dir;
}

// Position d'une commune : API Adresse (data.gouv, gratuite, sans clé) puis chef-lieu du département en secours
async function geocode(cp, ville) {
  cp = String(cp || '').replace(/\D/g, '').slice(0, 5);
  if (!/^\d{5}$/.test(cp)) return null;
  try {
    const q = (ville && String(ville).trim().length >= 3 ? String(ville).trim() : cp);
    const r = await fetch('https://api-adresse.data.gouv.fr/search/?q=' + encodeURIComponent(q) + '&postcode=' + cp + '&type=municipality&limit=1', { headers: { 'User-Agent': 'les-pieux-horizon-chatbot' } });
    const j = await r.json();
    const f = j && j.features && j.features[0];
    if (f && f.geometry) return { lat: f.geometry.coordinates[1], lon: f.geometry.coordinates[0], ville: f.properties && f.properties.city, approx: false };
  } catch (e) { console.error('geocode', String(e)); }
  const c = DEPT_CENTROIDS[cp.slice(0, 2)];
  return c ? { lat: c[0], lon: c[1], ville: ville || '', approx: true } : null;
}

// Adresse exacte du chantier : vérifiée et normalisée via la base adresse nationale
async function geocodeAddress(adresse, cp, ville) {
  const q = [adresse, cp, ville].filter(Boolean).join(' ').trim();
  if (!adresse || q.length < 6) return null;
  try {
    const r = await fetch('https://api-adresse.data.gouv.fr/search/?q=' + encodeURIComponent(q) + (cp && /^\d{5}$/.test(cp) ? '&postcode=' + cp : '') + '&limit=1', { headers: { 'User-Agent': 'les-pieux-horizon-chatbot' } });
    const j = await r.json();
    const f = j && j.features && j.features[0];
    if (!f || !f.geometry) return null;
    const pr = f.properties || {};
    const precise = ['housenumber', 'street', 'locality'].includes(pr.type) && (pr.score || 0) >= 0.45;
    return { lat: f.geometry.coordinates[1], lon: f.geometry.coordinates[0], label: pr.label || q, type: pr.type, score: pr.score || 0, precise, ville: pr.city || ville || '' };
  } catch (e) { console.error('geocodeAddress', String(e)); return null; }
}

function libelleFr(d) {
  const p = parisParts(d);
  const day = d.toLocaleDateString('fr-FR', { timeZone: 'Europe/Paris', weekday: 'long', day: 'numeric', month: 'long' });
  return day + ' à ' + p.h + ' h' + (p.mi ? ' ' + String(p.mi).padStart(2, '0') : '');
}

  const busy = [];
  for (const k of Object.keys(fb.calendars || {})) for (const b of (fb.calendars[k].busy || [])) busy.push([Date.parse(b.start), Date.parse(b.end)]);
  const ev = await gapi(env, 'https://www.googleapis.com/calendar/v3/calendars/' + encodeURIComponent(cal.visites) + '/events?singleEvents=true&maxResults=100&timeMin=' + encodeURIComponent(fromIso) + '&timeMax=' + encodeURIComponent(toIso) + '&fields=items(id,summary,start,end,extendedProperties)');
  const visits = (ev.items || []).filter(e => e.start && e.start.dateTime).map(e => { const pr = (e.extendedProperties && e.extendedProperties.private) || {}; return { s: Date.parse(e.start.dateTime), e: Date.parse(e.end.dateTime), secteur: pr.secteur, trajet: parseInt(pr.trajet, 10) || 0 }; });
  // Les visites existantes bloquent leur tampon de trajet ; une visite lointaine bloque tout son trajet pour les autres secteurs
  for (const v of visits) { const ext = (secteur && v.secteur === secteur) ? cfg.tampon : Math.max(cfg.tampon, v.trajet); busy.push([v.s - ext * 60000, v.e + ext * 60000]); }
  return { busy, visits };
}

// origine : 'chatbot' (garde-fou distance + accord déclaré) ou 'lien' (lien envoyé avec le chiffrage : Anthony a déjà décidé du déplacement)
async function agendaSlots(env, input, convId, origine) {
  const cfg = await agendaConfig(env);
  const cal = await agendaCalendars(env);
  if (!cal || !cal.visites) return { ok: false, raison: 'agenda non configuré : proposer un rappel téléphonique' };
  if (origine !== 'lien' && !input.accord_chiffrage) return { ok: false, raison: 'accord_chiffrage_requis', consigne: 'La visite terrain se planifie après acceptation du chiffrage estimatif. Expliquer le parcours : chiffrage estimatif gratuit sous 48 h à partir des informations, photos et plan, puis visite pour finaliser le devis, avec un lien de réservation dans l\'email du chiffrage. Ne proposer aucun créneau.' };
  await homeGeo(env);
  const geo = await geocode(input.code_postal, input.ville);
  if (!geo) return { ok: false, raison: 'code postal inconnu : demander le code postal du terrain' };
  const route = await routeMinutes(env, geo, input.code_postal);
  const trajet = route.min;
  if (env.CHAT_LOG && convId) { try { await env.CHAT_LOG.put('visite:conv:' + convId, JSON.stringify({ trajet_min: trajet, trajet_km: route.km, trajet_source: route.source, accord_chiffrage: !!input.accord_chiffrage, email: input.email || '', at: new Date().toISOString() }), { expirationTtl: LOG_TTL_DAYS * 86400 }); } catch (e) {} }
  if (origine !== 'lien' && trajet > cfg.trajet_max_min) {
    return { ok: false, raison: 'deplacement_long', trajet_min: trajet, consigne: 'Terrain trop éloigné pour une réservation par le chatbot : un conseiller rappelle sous 24 h pour fixer la visite. Ne pas parler de distance au visiteur.' };
  }
  const secteur = sectorOf(geo, input.code_postal);
  const now = new Date(), p = parisParts(now);
  const first = addDays(p.y, p.m, p.d, cfg.delai_min_jours);
  const last = addDays(p.y, p.m, p.d, cfg.horizon_jours);
  const fromIso = parisDate(first.y, first.m, first.d, 0, 0).toISOString();
  const toIso = parisDate(last.y, last.m, last.d, 23, 59).toISOString();
  const { busy, visits } = await agendaBusy(env, cal, fromIso, toIso, cfg, secteur);
  const [dh, dm] = hm(cfg.depart_min), [rh, rm] = hm(cfg.retour_max);
  const halfDays = [];
  for (let n = cfg.delai_min_jours; n <= cfg.horizon_jours; n++) {
    const day = addDays(p.y, p.m, p.d, n);
    const wd = new Date(Date.UTC(day.y, day.m - 1, day.d)).getUTCDay();
    if (!cfg.jours.includes(wd) || isHoliday(day.y, day.m, day.d)) continue;
    for (const win of [cfg.matin, cfg.apres_midi]) {
      if (!win || !win[0] || !win[1]) continue;
      const [h1, m1] = hm(win[0]), [h2, m2] = hm(win[1]);
      const ws = parisDate(day.y, day.m, day.d, h1, m1).getTime(), we = parisDate(day.y, day.m, day.d, h2, m2).getTime();
      // Départ du domicile au plus tôt à depart_min, retour au plus tard à retour_max (trajet réel compris)
      const tMin = parisDate(day.y, day.m, day.d, dh, dm).getTime() + trajet * 60000;
      const tMaxEnd = parisDate(day.y, day.m, day.d, rh, rm).getTime() - trajet * 60000;
      const inWin = visits.filter(v => overlaps(v.s, v.e, ws, we));
      if (inWin.length >= cfg.max_par_demi_journee) continue;
      if (inWin.some(v => v.secteur && v.secteur !== secteur)) continue; // demi-journée déjà dédiée à un autre secteur
      const same = inWin.length > 0;
      let bestT = null, bestDist = Infinity;
      for (let t = ws; t + cfg.duree * 60000 <= we; t += 30 * 60000) {
        if (t < tMin || t + cfg.duree * 60000 > tMaxEnd) continue;
        const s = t - cfg.tampon * 60000, e = t + (cfg.duree + cfg.tampon) * 60000;
        if (busy.some(b => overlaps(s, e, b[0], b[1]))) continue;
        if (!same) { bestT = t; break; }
        // Même secteur : on colle le nouveau créneau au plus près des visites déjà posées
        const dist = Math.min(...inWin.map(v => Math.min(Math.abs(t - v.e), Math.abs(v.s - (t + cfg.duree * 60000)))));
        if (dist < bestDist) { bestDist = dist; bestT = t; }
      }
      if (bestT === null) continue;
      halfDays.push({ t: bestT, same, day: day.y + '-' + day.m + '-' + day.d });
    }
  }
  halfDays.sort((a, b) => (a.same === b.same ? a.t - b.t : a.same ? -1 : 1));
  const chosen = [];
  for (const h of halfDays) { if (chosen.length >= 3) break; if (chosen.some(c => c.day === h.day)) continue; chosen.push(h); }
  for (const h of halfDays) { if (chosen.length >= 3) break; if (!chosen.includes(h)) chosen.push(h); }
  chosen.sort((a, b) => a.t - b.t);
  return {
    ok: true, secteur, trajet_min: trajet, trajet_km: route.km, trajet_source: route.source, commune: geo.ville || '',
    creneaux: chosen.map(h => ({ id: new Date(h.t).toISOString(), libelle: libelleFr(new Date(h.t)) })),
    consigne: chosen.length ? 'Proposer ces créneaux dans une seule phrase, sans en inventer d\'autres. Si aucun ne convient, proposer un rappel téléphonique.' : 'Aucun créneau libre sur les prochaines semaines : proposer un rappel téléphonique pour trouver une date.',
  };
}

function icsFor({ uid, start, end, summary, location, description }) {
  const f = d => new Date(d).toISOString().replace(/[-:]/g, '').replace(/\.\d{3}/, '');
  const esc = s => String(s || '').replace(/\\/g, '\\\\').replace(/\n/g, '\\n').replace(/,/g, '\\,').replace(/;/g, '\\;');
  return ['BEGIN:VCALENDAR', 'VERSION:2.0', 'PRODID:-//Les Pieux Horizon//Visite terrain//FR', 'METHOD:PUBLISH', 'BEGIN:VEVENT',
    'UID:' + uid, 'DTSTAMP:' + f(new Date()), 'DTSTART:' + f(start), 'DTEND:' + f(end), 'SUMMARY:' + esc(summary), 'LOCATION:' + esc(location), 'DESCRIPTION:' + esc(description),
    'END:VEVENT', 'END:VCALENDAR'].join('\r\n');
}

async function agendaBook(env, input, convId, origine) {
  const cfg = await agendaConfig(env);
  const cal = await agendaCalendars(env);
  if (!cal || !cal.visites) return { ok: false, raison: 'agenda non configuré : proposer un rappel téléphonique' };
  const start = new Date(String(input.debut || ''));
  if (isNaN(start.getTime())) return { ok: false, raison: 'créneau invalide : reproposer les créneaux avec creneaux_visite' };
  const p = parisParts(start), now = parisParts(new Date());
  const minDay = addDays(now.y, now.m, now.d, cfg.delai_min_jours);
  if (Date.UTC(p.y, p.m - 1, p.d) < Date.UTC(minDay.y, minDay.m - 1, minDay.d) || !cfg.jours.includes(p.wd)) return { ok: false, raison: 'créneau hors des jours de visite : reproposer les créneaux avec creneaux_visite' };
  if (!String(input.nom || '').trim() || !String(input.telephone || '').trim()) return { ok: false, raison: 'nom et téléphone requis avant de réserver' };
  if (!String(input.adresse || '').trim()) return { ok: false, raison: 'adresse_requise', consigne: 'Demander l\'adresse exacte du chantier : numéro et voie, ou lieu-dit, avec la commune.' };
  await homeGeo(env);
  const addr = await geocodeAddress(input.adresse, input.code_postal, input.ville);
  if (!addr || !addr.precise) return { ok: false, raison: 'adresse_introuvable', consigne: 'L\'adresse « ' + String(input.adresse).slice(0, 80) + ' » n\'est pas reconnue. Demander de la préciser : numéro et nom de la voie, ou nom exact du lieu-dit, et la commune. Ne pas réserver tant que l\'adresse n\'est pas précise.' };
  const geo = addr;
  const secteur = sectorOf(geo, input.code_postal);
  const route = await routeMinutes(env, geo, input.code_postal, true);
  const itineraire = 'https://www.google.com/maps/dir/?api=1&destination=' + geo.lat.toFixed(6) + ',' + geo.lon.toFixed(6);
  const end = new Date(start.getTime() + cfg.duree * 60000);
  // Vérification finale de disponibilité (le créneau a pu être pris entre-temps)
  const { busy } = await agendaBusy(env, cal, new Date(start.getTime() - 6 * 3600000).toISOString(), new Date(end.getTime() + 6 * 3600000).toISOString(), cfg, secteur);
  const s = start.getTime() - cfg.tampon * 60000, e = end.getTime() + cfg.tampon * 60000;
  if (busy.some(b => overlaps(s, e, b[0], b[1]))) return { ok: false, raison: 'créneau plus disponible : reproposer les créneaux avec creneaux_visite' };

  const who = [input.civilite, input.prenom, input.nom].filter(Boolean).join(' ');
  const ville = geo.ville || input.ville || '';
  const acces = String(input.acces || '').trim().slice(0, 200);
  const lieu = addr.label + (acces ? ' (' + acces + ')' : '');
  const libelle = libelleFr(start);
  const desc = ['Visite terrain réservée via le chatbot du site.', 'Client : ' + who, 'Téléphone : ' + input.telephone, input.email ? 'Email : ' + input.email : '',
    'Adresse du chantier : ' + addr.label + (acces ? '\nAccès : ' + acces : '') + '\nItinéraire : ' + itineraire,
    'Projet : ' + [input.type_projet, input.surface, input.terrain].filter(Boolean).join(', '), origine === 'lien' ? 'Réservée par le client via le lien du chiffrage estimatif (accord sur le chiffrage)' : 'Réservée via le chatbot : accord sur le chiffrage estimatif déclaré par le visiteur, à vérifier', 'Secteur : ' + secteur + (route ? ' (' + route.min + ' min de route' + (route.km ? ', ' + route.km + ' km' : '') + (route.source === 'maps' ? ', Google Maps' : ', estimation') + ' depuis ' + HOME_ADDRESS + ')' : ''), convId ? 'Conversation : ' + convId : ''].filter(Boolean).join('\n');
  const ev = await gapi(env, 'https://www.googleapis.com/calendar/v3/calendars/' + encodeURIComponent(cal.visites) + '/events?sendUpdates=none', 'POST', {
    summary: 'Visite terrain – ' + [input.prenom, input.nom].filter(Boolean).join(' ') + (input.type_projet ? ' – ' + input.type_projet : '') + (ville ? ' – ' + ville : ''),
    location: addr.label, description: desc,
    start: { dateTime: start.toISOString(), timeZone: 'Europe/Paris' }, end: { dateTime: end.toISOString(), timeZone: 'Europe/Paris' },
    attendees: input.email ? [{ email: input.email, displayName: who }] : [],
    extendedProperties: { private: { secteur, trajet: String(route ? route.min : ''), convId: convId || '', telephone: normPhone(input.telephone).replace(/^\+33/, '0') } },
    reminders: { useDefault: false, overrides: [{ method: 'popup', minutes: 18 * 60 }, { method: 'popup', minutes: 90 }] },
  });
  const rdv = { debut: start.toISOString(), fin: end.toISOString(), libelle, lieu, adresse: addr.label, acces, itineraire, secteur, eventId: ev.id, lien: ev.htmlLink || '', email: input.email || '', origine: origine || 'chatbot', reserve: new Date().toISOString() };
  if (env.CHAT_LOG && convId) await env.CHAT_LOG.put('rdv:conv:' + convId, JSON.stringify(rdv), { expirationTtl: LOG_TTL_DAYS * 86400 });

  let emailEnvoye = false;
  if (input.email && /^[\w.+-]+@[\w-]+\.[\w.-]+$/.test(input.email)) {
    try {
      const t = await getTemplates(env);
      const vars = { salutation: salutationFor(input.civilite, input.nom), civilite: input.civilite || '', nom: input.nom || '', prenom: input.prenom || '', date_visite: libelle, adresse: lieu, itineraire, projet: [input.type_projet, input.surface].filter(Boolean).join(', ') };
      const text = fillTemplate(t.visite_body, vars);
      const ics = icsFor({ uid: ev.id + '@les-pieux-horizon.com', start, end, summary: 'Visite terrain Les Pieux Horizon', location: addr.label, description: 'Visite terrain avec Les Pieux Horizon. Contact : 07 69 73 79 61.' });
      await gmailSend(env, { to: input.email, subject: fillTemplate(t.visite_subject, vars), text, html: brandedHtml(text), attachments: [{ name: 'visite-terrain.ics', mime: 'text/calendar; method=PUBLISH', content: ics }] });
      emailEnvoye = true;
    } catch (e) { console.error('email visite', String(e && e.message ? e.message : e)); }
  }
  return { ok: true, libelle, lieu, email_envoye: emailEnvoye, consigne: 'Confirmer la date et le lieu en une phrase' + (emailEnvoye ? ', dire qu\'une confirmation vient d\'être envoyée par email' : (input.email ? '' : ', proposer une adresse email pour recevoir la confirmation (une seule fois)')) + '. Terminer par le marqueur [LEAD] à jour.' };
}

}

// Après réservation via le lien : note HubSpot sur le contact (recherche par email puis téléphone), mise à jour de la transaction ouverte
async function visiteToHubspot(env, rec, r) {
  if (!env.HUBSPOT_TOKEN) return;
  const filterGroups = [];
  if (rec.email) filterGroups.push({ filters: [{ propertyName: 'email', operator: 'EQ', value: rec.email }] });
  if (rec.telephone) filterGroups.push({ filters: [{ propertyName: 'phone', operator: 'EQ', value: rec.telephone }] });
  if (!filterGroups.length) return;
  const found = await hs('/crm/v3/objects/contacts/search', 'POST', { filterGroups, properties: ['email'], limit: 1 }, env).catch(() => ({ results: [] }));
  const contact = (found.results || [])[0];
  if (!contact) return;
  const txt = 'VISITE TERRAIN : ' + r.libelle + ' — ' + r.lieu + '\nRéservée par le client via le lien du chiffrage estimatif (accord sur le chiffrage).';
  const assoc = [{ to: { id: contact.id }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 202 }] }];
  const deals = await hs('/crm/v3/objects/deals/search', 'POST', { filterGroups: [{ filters: [{ propertyName: 'associations.contact', operator: 'EQ', value: contact.id }] }], properties: ['dealname', 'dealstage', 'description'], limit: 5 }, env).catch(() => ({ results: [] }));
  const deal = (deals.results || []).find(d => !['closedwon', 'closedlost'].includes(d.properties.dealstage));
  if (deal) {
    assoc.push({ to: { id: deal.id }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 214 }] });
    await hs('/crm/v3/objects/deals/' + deal.id, 'PATCH', { properties: { description: txt + '\n\n' + (deal.properties.description || '') } }, env).catch(() => {});
  }
  await hs('/crm/v3/objects/notes', 'POST', { properties: { hs_timestamp: new Date().toISOString(), hs_note_body: esc(txt), hubspot_owner_id: HS_OWNER_ID }, associations: assoc }, env);
  await hs('/crm/v3/objects/tasks', 'POST', {
    properties: { hs_timestamp: nextWorkdayAt(10).toISOString(), hs_task_subject: 'Préparer la visite terrain du ' + r.libelle + ' — ' + [rec.prenom, rec.nom].filter(Boolean).join(' ') + (rec.type_projet ? // Création des agendas Visites, Chantiers, Matériel dans le compte Google connecté
async function agendaInit(env) {
  const ids = (await env.CHAT_LOG.get('google:calendars', 'json')) || {};
  const list = await gapi(env, 'https://www.googleapis.com/calendar/v3/users/me/calendarList?minAccessRole=owner&fields=items(id,summary)');
  for (const [key, name, color] of [['visites', 'Visites', '10'], ['chantiers', 'Chantiers', '6'], ['materiel', 'Matériel', '5']]) {
    if (ids[key]) continue;
    const ex = (list.items || []).find(c => c.summary === name);
    if (ex) { ids[key] = ex.id; continue; }
    const c = await gapi(env, 'https://www.googleapis.com/calendar/v3/calendars', 'POST', { summary: name, timeZone: 'Europe/Paris', description: 'Créé par le Worker Les Pieux Horizon' });
    ids[key] = c.id;
    await gapi(env, 'https://www.googleapis.com/calendar/v3/users/me/calendarList/' + encodeURIComponent(c.id), 'PATCH', { colorId: color }).catch(() => {});
  }
  ids.created = ids.created || new Date().toISOString();
  await env.CHAT_LOG.put('google:calendars', JSON.stringify(ids));
  return ids;
}

// Visites à venir (page /admin/agenda)
async function agendaUpcoming(env, days) {
  const cal = await agendaCalendars(env);
  if (!cal || !cal.visites) return [];
  const from = new Date().toISOString(), to = new Date(Date.now() + days * 86400000).toISOString();
  const ev = await gapi(env, 'https://www.googleapis.com/calendar/v3/calendars/' + encodeURIComponent(cal.visites) + '/events?singleEvents=true&orderBy=startTime&maxResults=50&timeMin=' + encodeURIComponent(from) + '&timeMax=' + encodeURIComponent(to) + '&fields=items(id,summary,location,start,end,htmlLink,extendedProperties)');
  return (ev.items || []).filter(e => e.start && e.start.dateTime).map(e => ({ id: e.id, summary: e.summary, location: e.location, start: e.start.dateTime, end: e.end.dateTime, link: e.htmlLink, secteur: e.extendedProperties && e.extendedProperties.private && e.extendedProperties.private.secteur }));
}

const HS = 'https://api.hubapi.com';
const HS_OWNER_ID = '956264949';        // Anthony Ouzeau
const HS_PIPELINE = 'default';          // Sales Pipeline
const HS_STAGE_NEW = 'appointmentscheduled'; // 1re étape (à renommer « Nouveau lead chatbot » dans HubSpot)

async function hs(path, method, body, env) {
  const res = await fetch(HS + path, {
    method,
    headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + env.HUBSPOT_TOKEN },
    body: body ? JSON.stringify(body) : undefined,
  });
  const txt = await res.text();
  let data = {};
  try { data = txt ? JSON.parse(txt) : {}; } catch (e) { data = { raw: txt }; }
  if (!res.ok) {
    console.error('Erreur HubSpot', method, path, res.status, txt.slice(0, 500));
    throw new Error('HubSpot ' + res.status + ' sur ' + path);
  }
  return data;
}

function normPhone(p) {
  return String(p || '').replace(/[^\d+]/g, '');
}

function esc(t) {
  return String(t || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/\n/g, '<br>');
}

function buildDealName(lead, label) {
  const parts = [label || 'Chatbot'];
  if (lead.type_projet) parts.push(lead.type_projet);
  if (lead.surface) parts.push(lead.surface);
  const who = [lead.prenom, lead.nom].filter(Boolean).join(' ');
  if (who) parts.push(who);
  if (lead.code_postal) parts.push(lead.code_postal);
  return parts.join(' – ');
}

function parseAmount(budget) {
  if (!budget) return null;
  const m = String(budget).replace(/\s/g, '').match(/(\d{3,7})/);
  return m ? m[1] : null;
}

async function rememberIds(env, convId, ids) {
  if (!env.CHAT_LOG || !convId) return;
  try {
    const prev = (await env.CHAT_LOG.get('hs:' + convId, 'json')) || {};
    await env.CHAT_LOG.put('hs:' + convId, JSON.stringify({ ...prev, ...ids }), { expirationTtl: LOG_TTL_DAYS * 86400 });
  } catch (e) { console.error('Erreur rememberIds', String(e)); }
}

async function syncHubspot(lead, transcript, env, convId, source, origin) {
  const isForm = source === 'form';
  const label = isForm ? 'Formulaire' : 'Chatbot';
  try {
    if (!env.HUBSPOT_TOKEN) { console.error('HUBSPOT_TOKEN manquant'); return; }
    const phone = normPhone(lead.telephone);
    const projet = [
      lead.type_projet, lead.surface, lead.terrain ? 'terrain ' + lead.terrain : '',
      lead.systeme, lead.echeance ? 'échéance ' + lead.echeance : '', lead.budget ? 'budget ' + lead.budget : '',
    ].filter(Boolean).join(', ');

    // --- Contact : recherche par téléphone, puis création ou mise à jour ---
    const contactProps = {
      firstname: lead.prenom || 'Prospect chatbot',
      phone: lead.telephone || '',
      zip: lead.code_postal || '',
      city: lead.ville || '',
      message: 'Projet : ' + (projet || 'non précisé') + (isForm ? ' — via formulaire site' : ' — via chatbot site'),
      hs_lead_status: 'NEW',
      lifecyclestage: 'lead',
      hubspot_owner_id: HS_OWNER_ID,
    };
    if (lead.nom) contactProps.lastname = lead.nom;
    if (lead.civilite) contactProps.salutation = lead.civilite;
    if (lead.email) contactProps.email = lead.email;

    // Ids déjà connus pour cette conversation (évite les doublons quand le visiteur
    // complète ses coordonnées avant que la recherche HubSpot ait indexé le contact)
    let known = null;
    if (env.CHAT_LOG && convId) {
      try { known = await env.CHAT_LOG.get('hs:' + convId, 'json'); } catch (e) { known = null; }
    }

    let contactId = known && known.contactId ? String(known.contactId) : null;
    const filterGroups = [];
    if (lead.telephone) filterGroups.push({ filters: [{ propertyName: 'phone', operator: 'EQ', value: lead.telephone }] });
    if (lead.email) filterGroups.push({ filters: [{ propertyName: 'email', operator: 'EQ', value: lead.email }] });
    if (!contactId && filterGroups.length) {
      const found = await hs('/crm/v3/objects/contacts/search', 'POST', {
        filterGroups, properties: ['phone', 'email'], limit: 5,
      }, env);
      const existing = (found.results || []).find(c => phone && normPhone(c.properties.phone) === phone)
        || (found.results || []).find(c => lead.email && (c.properties.email || '').toLowerCase() === lead.email.toLowerCase())
        || (found.results || [])[0];
      if (existing) contactId = existing.id;
    }
    if (contactId) {
      delete contactProps.hs_lead_status; delete contactProps.lifecyclestage; // on ne rétrograde pas un contact déjà travaillé
      if (!lead.prenom) delete contactProps.firstname;
      if (known && known.reconnu && !lead.type_projet) delete contactProps.message; // client reconnu : on ne remplace pas la fiche projet par du vide
      await hs('/crm/v3/objects/contacts/' + contactId, 'PATCH', { properties: contactProps }, env);
    } else {
      const created = await hs('/crm/v3/objects/contacts', 'POST', { properties: contactProps }, env);
      contactId = created.id;
    }
    await rememberIds(env, convId, { contactId });

    // Visite terrain réservée par le chatbot pendant cette conversation ?
    let rdv = null, depl = null;
    if (env.CHAT_LOG && convId) { try { rdv = await env.CHAT_LOG.get('rdv:conv:' + convId, 'json'); depl = await env.CHAT_LOG.get('visite:conv:' + convId, 'json'); } catch (e) { rdv = null; } }

    // --- Transaction : une seule par contact venant du chatbot, mise à jour si elle existe ---
    const fiche =
      'Type : ' + (lead.type_projet || '-') + '\nSurface : ' + (lead.surface || '-') + (lead.forme || lead.longueur ? '\nForme : ' + [lead.forme, lead.longueur && lead.largeur ? lead.longueur + ' x ' + lead.largeur + ' m' : ''].filter(Boolean).join(' ') : '') +
      (lead.platelage ? '\nPlatelage : ' + lead.platelage : '') + (lead.hauteur ? '\nHauteur du plancher : ' + lead.hauteur + ' m' : '') + (lead.adossee ? '\nAdossée : ' + lead.adossee : '') + (lead.charges ? '\nCharges : ' + lead.charges : '') + (lead.profil ? '\nProfil : ' + lead.profil : '') + (lead.usage ? '\nUsage : ' + lead.usage : '') +
      '\nTerrain : ' + (lead.terrain || '-') + '\nSystème : ' + (lead.systeme || '-') +
      '\nLieu : ' + [lead.code_postal, lead.ville].filter(Boolean).join(' ') +
      '\nÉchéance : ' + (lead.echeance || '-') + '\nBudget annoncé : ' + (lead.budget || '-') +
      '\nSource : ' + (isForm ? 'formulaire' : 'chatbot') + ' les-pieux-horizon.com' + (convId ? '\nID conversation : ' + convId : '');
    const deplTxt = depl ? 'Déplacement : ' + depl.trajet_min + ' min' + (depl.trajet_km ? ' (' + depl.trajet_km + ' km)' : '') + ' depuis le domicile' + (depl.accord_chiffrage ? ' — accord sur le chiffrage estimatif DÉCLARÉ par le visiteur dans le chatbot, à vérifier' : '') : '';
    const rdvTxt = (rdv ? 'VISITE TERRAIN : ' + rdv.libelle + ' — ' + rdv.lieu + (rdv.lien ? '\n' + rdv.lien : '') : '') + (deplTxt ? (rdv ? '\n' : '') + deplTxt : '');
    const description = (rdvTxt ? rdvTxt + '\n\n' : '') + (lead.resume ? lead.resume + '\n\n' : '') + fiche;
    const dealProps = {
      dealname: buildDealName(lead, label),
      description,
      hubspot_owner_id: HS_OWNER_ID,
      dealtype: 'newbusiness',
    };
    const amount = parseAmount(lead.budget);
    if (amount) dealProps.amount = amount;

    let dealId = known && known.dealId ? String(known.dealId) : null;
    const deals = dealId ? { results: [] } : await hs('/crm/v3/objects/deals/search', 'POST', {
      filterGroups: [{ filters: [{ propertyName: 'associations.contact', operator: 'EQ', value: contactId }] }],
      properties: ['dealname', 'dealstage'], sorts: [{ propertyName: 'hs_lastmodifieddate', direction: 'DESCENDING' }], limit: 5,
    }, env).catch(() => ({ results: [] }));
    const openDeal = (deals.results || []).find(d => !['closedwon', 'closedlost'].includes(d.properties.dealstage));
    if (!dealId && openDeal) dealId = openDeal.id;
    const isUpdate = !!dealId;
    if (dealId) {
      if (openDeal && openDeal.properties.dealname && !openDeal.properties.dealname.startsWith(label)) delete dealProps.dealname;
      await hs('/crm/v3/objects/deals/' + dealId, 'PATCH', { properties: dealProps }, env);
    } else {
      dealProps.pipeline = HS_PIPELINE;
      dealProps.dealstage = HS_STAGE_NEW;
      const created = await hs('/crm/v3/objects/deals', 'POST', {
        properties: dealProps,
        associations: [{ to: { id: contactId }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 3 }] }],
      }, env);
      dealId = created.id;
    }
    await rememberIds(env, convId, { contactId, dealId });

    // --- Note : compte rendu + conversation complète ---
    const convo = transcript
      .map(m => (m.role === 'user' ? '<b>Visiteur :</b> ' : '<b>Fernand :</b> ') + esc(String(m.content || '').replace(/\[LEAD\][\s\S]*$/, '').trim()))
      .join('<br><br>');
    const noteBody =
      '<b>' + (isForm ? 'Demande via formulaire du site' : (known && known.reconnu ? 'Client revenu sur le chatbot' : 'Compte rendu chatbot')) + '' + '</b><br>' + esc(lead.resume || projet || 'Aucun résumé') +
      (rdvTxt ? '<br><br><b>' + (rdv ? 'Visite terrain réservée' : 'Visite terrain') + '</b><br>' + esc(rdvTxt) : '') +
      '<br><br><b>Fiche projet</b><br>' + esc(fiche) +
      '<br><br><b>' + (isForm ? 'Données envoyées' : 'Conversation') + '</b><br>' + convo;
    const assoc = [{ to: { id: contactId }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 202 }] }];
    if (dealId) assoc.push({ to: { id: dealId }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 214 }] });
    // Pièces reçues par le chatbot (dépôt de fichiers)
    let filesTxt = '';
    try { const fl = convId && env.CHAT_LOG ? await env.CHAT_LOG.get('files:' + convId, 'json') : null; if (fl && fl.length) filesTxt = '<br><br><b>Pièces reçues</b><br>' + fl.map(f => '<a href="' + esc(f.link) + '">' + esc(f.name) + '</a>').join('<br>'); } catch (e) {}
    // Une seule note par conversation : la note existante est mise à jour, sauf pour un client reconnu qui revient (nouvelle note)
    if (known && known.noteId && !(known && known.reconnu)) {
      await hs('/crm/v3/objects/notes/' + known.noteId, 'PATCH', { properties: { hs_note_body: (noteBody + filesTxt).slice(0, 60000) } }, env);
    } else {
      const note = await hs('/crm/v3/objects/notes', 'POST', {
        properties: { hs_timestamp: new Date().toISOString(), hs_note_body: (noteBody + filesTxt).slice(0, 60000), hubspot_owner_id: HS_OWNER_ID },
        associations: assoc,
      }, env);
      if (note && note.id) await rememberIds(env, convId, { noteId: note.id });
    }

    console.log('HubSpot OK contact', contactId, 'deal', dealId);

    // --- Automatisations après lead (chacune une seule fois par contact) ---
    await afterLead(lead, contactId, dealId, env, isForm, rdv, depl, convId, origin);
  } catch (e) {
    console.error('Erreur syncHubspot', String(e && e.message ? e.message : e));
  }
}

// Une seule fois par contact et par action : dossier Drive, accusé de réception, tâche de rappel
async function afterLead(lead, contactId, dealId, env, isForm, rdv, depl, convId, origin) {
  const once = async (key, fn) => {
    if (!env.CHAT_LOG) return fn();
    const k = 'done:' + key + ':' + contactId;
    const prev = await env.CHAT_LOG.get(k);
    if (prev) return prev.startsWith('{') ? JSON.parse(prev) : null;
    const r = await fn();
    await env.CHAT_LOG.put(k, r && typeof r === 'object' && r.remember ? JSON.stringify(r) : new Date().toISOString(), { expirationTtl: LOG_TTL_DAYS * 86400 });
    return r;
  };

  // Dossier client dans le Drive (si Google est connecté et l'arborescence créée)
  let driveUrl = null;
  try {
    driveUrl = await once('drive', async () => {
      const tree = env.CHAT_LOG ? await env.CHAT_LOG.get('google:drive', 'json') : null;
      if (!tree || !tree.clients) return null;
      const name = [lead.code_postal, [lead.prenom, lead.nom].filter(Boolean).join(' '), lead.type_projet, 'LPH'].filter(Boolean).join(' – ');
      const folder = await driveFolder(env, name, tree.clients);
      for (const sub of ['Plans', 'Devis', 'Photos chantier', 'Rapport de pose', 'SAV']) await driveFolder(env, sub, folder.id);
      const link = 'https://drive.google.com/drive/folders/' + folder.id;
      if (dealId) {
        await hs('/crm/v3/objects/deals/' + dealId, 'PATCH', { properties: { description: (await hs('/crm/v3/objects/deals/' + dealId + '?properties=description', 'GET', null, env)).properties.description + '\nDossier Drive : ' + link } }, env).catch(() => {});
      }
      // Pièces déposées dans le chatbot avant la création du dossier : on les range dans Plans
      try {
        const fl = convId ? await env.CHAT_LOG.get('files:' + convId, 'json') : null;
        if (fl && fl.length) {
          const plans = await driveFolder(env, 'Plans', folder.id);
          for (const f of fl.slice(0, 6)) {
            const fid = (String(f.link || '').match(/\/d\/([\w-]+)/) || [])[1];
            if (!fid) continue;
            const cur = await gapi(env, 'https://www.googleapis.com/drive/v3/files/' + fid + '?fields=parents').catch(() => null);
            if (cur && cur.parents && !cur.parents.includes(plans.id)) await gapi(env, 'https://www.googleapis.com/drive/v3/files/' + fid + '?addParents=' + plans.id + '&removeParents=' + cur.parents.join(',') + '&fields=id', 'PATCH', { name: f.name }).catch(() => {});
          }
        }
      } catch (e) { console.error('rangement pièces chatbot', String(e)); }
      return { remember: true, folderId: folder.id, link };
    });
    if (driveUrl && driveUrl.link) driveUrl = driveUrl.link;
  } catch (e) { console.error('Drive dossier client', String(e && e.message ? e.message : e)); }

  // Accusé de réception au prospect
  if (lead.email) {
    try {
      await once('ack', () => sendAck(env, lead, isForm));
    } catch (e) { console.error('Accusé de réception', String(e && e.message ? e.message : e)); }
  }

  // Chiffrage estimatif automatique (terrasse de particulier, dossier complet) : une seule fois par contact
  try {
    // Pas de « once » ici : un dossier incomplet au premier passage (email, hauteur) doit pouvoir être chiffré au lead suivant ; seul un envoi réussi est mémorisé
    const kAuto = 'done:chiffrage:' + contactId;
    const prevAuto = env.CHAT_LOG ? await env.CHAT_LOG.get(kAuto) : null;
    const auto = prevAuto ? null : await chiffrageAuto(env, lead, origin);
    if (auto && auto.sent && env.CHAT_LOG) await env.CHAT_LOG.put(kAuto, JSON.stringify(auto), { expirationTtl: LOG_TTL_DAYS * 86400 });
    if (auto && auto.skipped && dealId && !auto.noted) {
      const why = { profil: 'profil non particulier', famille: 'projet hors terrasse', email: 'email manquant', dimensions: 'dimensions manquantes', hauteur: 'hauteur du plancher manquante', grille: 'grille tarifaire absente', etude: 'étude : ' + (auto.raison || ''), manquants: 'valeurs manquantes dans la grille : ' + (auto.raison || '') }[auto.skipped] || auto.skipped;
      const cur = await hs('/crm/v3/objects/deals/' + dealId + '?properties=description', 'GET', null, env).catch(() => null);
      if (cur && !/Chiffrage automatique non envoyé/.test(cur.properties.description || '')) await hs('/crm/v3/objects/deals/' + dealId, 'PATCH', { properties: { description: 'Chiffrage automatique non envoyé (' + why + ') : à faire à la main depuis le journal.\n' + (cur.properties.description || '') } }, env).catch(() => {});
    }
  } catch (e) { console.error('Chiffrage automatique', String(e && e.message ? e.message : e)); }

  // Tâche de rappel pour Anthony, échéance le prochain jour ouvré à 10 h
  try {
    await once('task', async () => {
      const due = nextWorkdayAt(10);
      const who = [lead.prenom, lead.nom].filter(Boolean).join(' ') || 'prospect';
      const body = 'Rappeler ' + who + (lead.telephone ? ' au ' + lead.telephone : '') + (lead.email ? ' (' + lead.email + ')' : '') +
        '\nProjet : ' + [lead.type_projet, lead.surface, lead.terrain, lead.code_postal].filter(Boolean).join(', ') +
        (lead.echeance ? '\nÉchéance : ' + lead.echeance : '') + (lead.budget ? '\nBudget annoncé : ' + lead.budget : '') +
        (rdv ? '\n\nVisite terrain réservée : ' + rdv.libelle + ' — ' + rdv.lieu + (rdv.lien ? '\n' + rdv.lien : '') : '') +
        (depl ? '\n\nDéplacement : ' + depl.trajet_min + ' min' + (depl.trajet_km ? ' (' + depl.trajet_km + ' km)' : '') + (depl.accord_chiffrage ? ' — accord chiffrage déclaré par le visiteur, à vérifier' : '') : '') +
        (lead.resume ? '\n\n' + lead.resume : '') + (driveUrl ? '\n\nDossier Drive : ' + driveUrl : '');
      const assoc = [{ to: { id: contactId }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 204 }] }];
      if (dealId) assoc.push({ to: { id: dealId }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 216 }] });
      const subject = rdv ? 'Préparer la visite terrain du ' + rdv.libelle + ' — ' + who + (lead.type_projet ? ' — ' + lead.type_projet : '')
        : 'Rappeler ' + who + ' — ' + (lead.type_projet || 'projet') + (lead.code_postal ? ' (' + lead.code_postal + ')' : '');
      return hs('/crm/v3/objects/tasks', 'POST', {
        properties: { hs_timestamp: due.toISOString(), hs_task_subject: subject,
          hs_task_body: body, hs_task_status: 'NOT_STARTED', hs_task_priority: 'HIGH', hs_task_type: 'CALL', hubspot_owner_id: HS_OWNER_ID },
        associations: assoc,
      }, env);
    });
  } catch (e) { console.error('Tâche de rappel', String(e && e.message ? e.message : e)); }
}

function nextWorkdayAt(hourParis) {
  const d = new Date();
  d.setUTCHours(hourParis - 2, 0, 0, 0); // Europe/Paris = UTC+2 en été, +1 en hiver (approximation acceptable pour une échéance)
  do { d.setUTCDate(d.getUTCDate() + 1); } while (d.getUTCDay() === 0 || d.getUTCDay() === 6);
  return d;
}

// « Bonjour M. Dupont, » / « Bonjour Mme Dupont, » ; sans civilité ni nom sûrs : « Bonjour, »
function salutationFor(civilite, nom) {
  const c = String(civilite || '').trim().replace(/^monsieur$/i, 'M.').replace(/^madame$/i, 'Mme').replace(/^mr\.?$/i, 'M.');
  const n = String(nom || '').trim();
  if (n && /^(M\.|Mme)$/.test(c)) return 'Bonjour ' + c + ' ' + n + ',';
  if (n) return 'Bonjour Madame, Monsieur ' + n + ',';
  return 'Bonjour,';
}

function workerOrigin(url) { return url.origin; }

async function oauthStart(url, env) {
  const token = url.searchParams.get('token') || '';
  if (!env.ADMIN_TOKEN || token !== env.ADMIN_TOKEN) return new Response('Accès refusé', { status: 403 });
  if (!env.GOOGLE_CLIENT_ID || !env.GOOGLE_CLIENT_SECRET) return new Response('GOOGLE_CLIENT_ID / GOOGLE_CLIENT_SECRET manquants dans les secrets du Worker', { status: 500 });
  const state = crypto.randomUUID();
  await env.CHAT_LOG.put('oauth:state:' + state, '1', { expirationTtl: 600 });
  const p = new URLSearchParams({
    client_id: env.GOOGLE_CLIENT_ID,
    redirect_uri: workerOrigin(url) + '/oauth/callback',
    response_type: 'code',
    scope: GOOGLE_SCOPES,
    access_type: 'offline',
    prompt: 'consent',
    include_granted_scopes: 'true',
    state,
  });
  return Response.redirect('https://accounts.google.com/o/oauth2/v2/auth?' + p.toString(), 302);
}

async function oauthCallback(url, env) {
  const state = url.searchParams.get('state') || '';
  const code = url.searchParams.get('code') || '';
  const err = url.searchParams.get('error');
  if (err) return new Response('Google a refusé : ' + err, { status: 400 });
  if (!state || !(await env.CHAT_LOG.get('oauth:state:' + state))) return new Response('État OAuth invalide ou expiré, recommencez depuis /admin/google', { status: 400 });
  await env.CHAT_LOG.delete('oauth:state:' + state);
  const res = await fetch('https://oauth2.googleapis.com/token', {
    method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({ code, client_id: env.GOOGLE_CLIENT_ID, client_secret: env.GOOGLE_CLIENT_SECRET, redirect_uri: workerOrigin(url) + '/oauth/callback', grant_type: 'authorization_code' }),
  });
  const data = await res.json();
  if (!res.ok || !data.refresh_token) return new Response('Échange de code échoué : ' + JSON.stringify(data).slice(0, 300) + '. Retirez l\'accès de l\'application dans myaccount.google.com/permissions puis recommencez.', { status: 500 });
  let email = '';
  try {
    const u = await fetch('https://openidconnect.googleapis.com/v1/userinfo', { headers: { Authorization: 'Bearer ' + data.access_token } });
    email = (await u.json()).email || '';
  } catch (e) {}
  await env.CHAT_LOG.put('google:tokens', JSON.stringify({ refresh_token: data.refresh_token, email, connected: new Date().toISOString() }));
  await env.CHAT_LOG.put('google:access', JSON.stringify({ access_token: data.access_token, exp: Date.now() + (data.expires_in - 60) * 1000 }), { expirationTtl: data.expires_in });
  return new Response('<!doctype html><meta charset="utf-8"><p style="font:15px Inter,system-ui;padding:32px">Google connecté : <b>' + escHtml(email) + '</b>. Vous pouvez fermer cet onglet et revenir sur la page /admin/google.</p>', { headers: { 'Content-Type': 'text/html; charset=utf-8' } });
}

async function googleToken(env) {
  if (!env.CHAT_LOG) throw new Error('KV absent');
  const cached = await env.CHAT_LOG.get('google:access', 'json');
  if (cached && cached.exp > Date.now()) return cached.access_token;
  const t = await env.CHAT_LOG.get('google:tokens', 'json');
  if (!t || !t.refresh_token) throw new Error('Google non connecté (voir /admin/google)');
  const res = await fetch('https://oauth2.googleapis.com/token', {
    method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({ refresh_token: t.refresh_token, client_id: env.GOOGLE_CLIENT_ID, client_secret: env.GOOGLE_CLIENT_SECRET, grant_type: 'refresh_token' }),
  });
  const data = await res.json();
  if (!res.ok || !data.access_token) throw new Error('Rafraîchissement du jeton Google échoué : ' + JSON.stringify(data).slice(0, 200));
  await env.CHAT_LOG.put('google:access', JSON.stringify({ access_token: data.access_token, exp: Date.now() + (data.expires_in - 60) * 1000 }), { expirationTtl: data.expires_in });
  return data.access_token;
}

async function gapi(env, url, method, body) {
  const tok = await googleToken(env);
  const res = await fetch(url, { method: method || 'GET', headers: { Authorization: 'Bearer ' + tok, 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined });
  const txt = await res.text();
  let data = {}; try { data = txt ? JSON.parse(txt) : {}; } catch (e) { data = { raw: txt }; }
  if (!res.ok) { console.error('Erreur Google', method || 'GET', url, res.status, txt.slice(0, 400)); throw new Error('Google ' + res.status + ' sur ' + url.replace('https://www.googleapis.com', '')); }
  return data;
}

function b64url(str) {
  const bytes = new TextEncoder().encode(str);
  let bin = ''; for (const b of bytes) bin += String.fromCharCode(b);
  return btoa(bin).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}
function mimeWord(s) { return '=?UTF-8?B?' + btoa(unescape(encodeURIComponent(s))) + '?='; }

async function gmailSend(env, { to, subject, text, html, attachments, cc }) {
  const from = env.MAIL_FROM || 'contact@chalets-horizon.com';
  const fromName = env.MAIL_FROM_NAME || 'Les Pieux Horizon';
  const b64 = s => btoa(unescape(encodeURIComponent(s)));
  const alt = 'lph' + Date.now().toString(36);
  const altPart = [
    '--' + alt, 'Content-Type: text/plain; charset="UTF-8"', 'Content-Transfer-Encoding: base64', '', b64(text),
    '--' + alt, 'Content-Type: text/html; charset="UTF-8"', 'Content-Transfer-Encoding: base64', '',
    b64(html || ('<div style="font:15px/1.6 Inter,Arial,sans-serif;color:#17251C;max-width:600px">' + escHtml(text).replace(/\n/g, '<br>') + '</div>')),
    '--' + alt + '--',
  ];
  const head = ['From: ' + mimeWord(fromName) + ' <' + from + '>', 'To: ' + to, ...(cc ? ['Cc: ' + cc] : []), 'Subject: ' + mimeWord(subject), 'MIME-Version: 1.0'];
  let lines;
  if (attachments && attachments.length) {
    const mixed = 'lphm' + Date.now().toString(36);
    lines = [...head, 'Content-Type: multipart/mixed; boundary="' + mixed + '"', '', '--' + mixed, 'Content-Type: multipart/alternative; boundary="' + alt + '"', '', ...altPart];
    for (const a of attachments) lines.push('--' + mixed, 'Content-Type: ' + (a.mime || 'application/octet-stream') + '; name="' + a.name + '"', 'Content-Disposition: attachment; filename="' + a.name + '"', 'Content-Transfer-Encoding: base64', '', b64(a.content));
    lines.push('--' + mixed + '--');
  } else {
    lines = [...head, 'Content-Type: multipart/alternative; boundary="' + alt + '"', '', ...altPart];
  }
  return gapi(env, 'https://gmail.googleapis.com/gmail/v1/users/me/messages/send', 'POST', { raw: b64url(lines.join('\r\n')) });
}

async function driveFolder(env, name, parentId) {
  const q = "name = '" + name.replace(/'/g, "\\'") + "' and mimeType = 'application/vnd.google-apps.folder' and trashed = false" + (parentId ? " and '" + parentId + "' in parents" : '');
  const found = await gapi(env, 'https://www.googleapis.com/drive/v3/files?q=' + encodeURIComponent(q) + '&fields=files(id,name)&pageSize=5');

  return { ...f, created: true };
}

// Création reprenable (limite Cloudflare : 50 appels réseau par exécution) :
// l'avancement est mémorisé dans KV après chaque dossier ; un nouvel appel reprend où il s'est arrêté.
async function driveInit(env) {
  const ids = (await env.CHAT_LOG.get('google:drive', 'json')) || {};
  const save = () => env.CHAT_LOG.put('google:drive', JSON.stringify(ids));
  let calls = 0;
  const budget = 36; // marge sous la limite de 50 (jeton Google + KV compris)
  const need = n => { calls += n; if (calls > budget) throw new Error('BUDGET'); };
  try {
    if (!ids.root) { need(2); const r = await driveFolder(env, 'Chalets Horizon', null); ids.root = r.id; ids.rootNew = !!r.created; await save(); }
    ids.doneSubs = ids.doneSubs || [];
    for (const [name, subs] of DRIVE_TREE) {
      const key = name.slice(0, 2);
      if (!ids[key]) {
        need(ids.rootNew ? 1 : 2);
        const f = ids.rootNew ? await driveCreateFolder(env, name, ids.root) : await driveFolder(env, name, ids.root);
        ids[key] = f.id; ids['new' + key] = ids.rootNew || !!f.created;
        if (name === '03 Clients') ids.clients = f.id;
        await save();
      }
      if (!ids.doneSubs.includes(key)) {
        for (const sub of subs) {
          const k2 = key + '/' + sub;
          if (ids[k2]) continue;
          need(ids['new' + key] ? 1 : 2);
          const f = ids['new' + key] ? await driveCreateFolder(env, sub, ids[key]) : await driveFolder(env, sub, ids[key]);
          ids[k2] = f.id; await save();
        }
        ids.doneSubs.push(key); await save();
      }
    }
    ids.created = ids.created || new Date().toISOString();
    ids.complete = true;
    await save();
    return ids;
  } catch (e) {
    if (String(e && e.message) === 'BUDGET') { ids.partial = true; await save(); return ids; }
    throw e;
  }
}

// ---------- Relève de la boîte Gmail : réponses des prospects → note HubSpot + pièces dans le Drive ----------
function b64urlDecode(str) {
  const b = atob(String(str || '').replace(/-/g, '+').replace(/_/g, '/'));
  const out = new Uint8Array(b.length);
  for (let i = 0; i < b.length; i++) out[i] = b.charCodeAt(i);
  return out;
}
function decodeText(data) { try { return new TextDecoder().decode(b64urlDecode(data)); } catch (e) { return ''; } }

function walkParts(payload, acc) {
  if (!payload) return acc;
  const mt = payload.mimeType || '';
  if (payload.filename && payload.body && payload.body.attachmentId) acc.files.push({ name: payload.filename, mime: mt, id: payload.body.attachmentId, size: payload.body.size || 0 });
  else if (mt === 'text/plain' && payload.body && payload.body.data && !acc.text) acc.text = decodeText(payload.body.data);
  else if (mt === 'text/html' && payload.body && payload.body.data && !acc.html) acc.html = decodeText(payload.body.data);
  for (const p of payload.parts || []) walkParts(p, acc);
  return acc;
}

async function driveUpload(env, name, mime, bytes, parentId) {
  const boundary = 'lphu' + Date.now().toString(36);
  const meta = JSON.stringify({ name, parents: [parentId] });
  const head = new TextEncoder().encode('--' + boundary + '\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n' + meta + '\r\n--' + boundary + '\r\nContent-Type: ' + (mime || 'application/octet-stream') + '\r\n\r\n');
  const tail = new TextEncoder().encode('\r\n--' + boundary + '--');
  const body = new Uint8Array(head.length + bytes.length + tail.length);
  body.set(head, 0); body.set(bytes, head.length); body.set(tail, head.length + bytes.length);
  const tok = await googleToken(env);
  const res = await fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id,name,webViewLink', {
    method: 'POST', headers: { Authorization: 'Bearer ' + tok, 'Content-Type': 'multipart/related; boundary=' + boundary }, body,
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error('Drive upload ' + res.status + ' ' + JSON.stringify(data).slice(0, 200));
  return data;
}

        const vars = { salutation: salutationFor(contact.properties.salutation, contact.properties.lastname), nom: contact.properties.lastname || '', prenom: contact.properties.firstname || '', fichiers_phrase: ' et ' + saved.length + ' pièce(s) jointe(s) : ' + saved.map(x => x.name).join(', '), nb_fichiers: String(saved.length), fichiers_liste: saved.map(x => '- ' + x.name).join('\n') };
        const body = fillTemplate(tpl.docs_body, vars);
        await gmailSend(env, { to: fromEmail, subject: fillTemplate(tpl.docs_subject, vars), text: body, html: brandedHtml(body) });
      } catch (e) { console.error('inbox confirmation', String(e)); }
    }
    await env.CHAT_LOG.put(key, 'consigned', { expirationTtl: 30 * 86400 });
    report.consigned++;
  }
  await env.CHAT_LOG.put('inbox:last', JSON.stringify({ at: new Date().toISOString(), ...report }));
  return report;
}

const num = v => { if (v === null || v === undefined || v === '') return null; const n = parseFloat(String(v).replace(',', '.').replace(/[^\d.-]/g, '')); return isNaN(n) ? null : n; };
const r2 = x => Math.round(x * 100) / 100;

// Prix unitaire selon la quantité : paliers triés par quantité ; sous le plus petit palier, prix du plus petit palier (sachet / pièce)
function prixPalier(item, qte) {
  const tiers = [];
  const baseQ = (item.conditionnement && !/^pi[eè]ce$/i.test(item.conditionnement) && num(item.pieces)) ? num(item.pieces) : 1;
  if (num(item.prix) !== null) tiers.push({ q: baseQ, p: num(item.prix) });
  if (num(item.q2) !== null && num(item.p2) !== null) tiers.push({ q: num(item.q2), p: num(item.p2) });
  if (num(item.q3) !== null && num(item.p3) !== null) tiers.push({ q: num(item.q3), p: num(item.p3) });
  if (!tiers.length) return null;
  tiers.sort((a, b) => a.q - b.q);
  let pu = tiers[0].p;
  for (const t of tiers) if (qte >= t.q) pu = t.p;
  return pu;
}

// Portée admissible d'une pièce de bois (Eurocode 5 simplifié) : flexion, flèche L/ratio, cisaillement
// b, h en mm ; trib = largeur reprise en m ; charges en kg/m² ; renvoie la portée en m
function porteeBois(b, h, trib, R) {
  const fmk = num(R.bois_fm_k) || 24, E = num(R.bois_E) || 11000, kmod = num(R.bois_kmod) || 0.8, gM = num(R.bois_gamma_m) || 1.3, fvk = num(R.bois_fv_k) || 4;
  const ratio = num(R.fleche_ratio) || 300, secu = num(R.coef_securite_portee) || 0.9, pmax = num(R.portee_max) || 3;
  const G = (num(R.charge_permanente) || 50) * 9.81 / 1000, Q = (num(R.charge_prive) || 250) * 9.81 / 1000; // kN/m²
  const wU = (1.35 * G + 1.5 * Q) * trib, wS = (G + Q) * trib; // kN/m
  const fmd = fmk * kmod / gM, fvd = fvk * kmod / gM;
  const W = b * h * h / 6, I = b * h * h * h / 12; // mm³, mm⁴
  // unités : N, mm ; 1 kN/m = 1 N/mm
  const Lflex = Math.sqrt(8 * fmd * W / wU) / 1000;                     // M = wL²/8 ≤ fmd·W
  const Lfle = Math.cbrt(384 * E * I / (5 * ratio * wS)) / 1000;        // 5wL⁴/384EI ≤ L/ratio
  const Lcis = 2 * (2 / 3) * fvd * b * h / wU / 1000;                   // V = wL/2 ≤ fvd·(2/3)·b·h
  return r2(Math.min(pmax, secu * Math.min(Lflex, Lfle, Lcis)));
}

function geometrie(input) {
  const L = num(input.longueur), l = num(input.largeur), L2 = num(input.longueur2), l2 = num(input.largeur2);
  let S = num(input.surface), P = num(input.perimetre), Pint = 0;
  const forme = String(input.forme || (L && l ? 'rectangle' : 'libre')).toLowerCase();
  if (forme === 'rectangle' && L && l) { S = L * l; P = 2 * (L + l); }
  else if (forme === 'l' && L && l && L2 && l2) { S = L * l + L2 * l2; P = 2 * (L + l) + 2 * L2; }
  else if (/piscine|spa|bassin|pourtour/.test(forme) && L && l && L2 && l2) { S = L * l - L2 * l2; P = 2 * (L + l); Pint = 2 * (L2 + l2); }
  if (!S) return null;
  if (!P) P = 4 * Math.sqrt(S) * 1.1; // forme inconnue : périmètre d'un carré majoré
  return { S: r2(S), P: r2(P), Pint: r2(Pint), forme };
}

  } else if (unNiveau) {
    // Forme non rectangulaire à un niveau : cadres en bandes de b m ; vis sous les jonctions (tous les a) et sous les rives (tous les aR)
    const spR = (trib, section) => { let x = porteeBois(section, 145, trib, R); if (x * trib > surfMaxVis) x = r2(Math.floor(surfMaxVis / trib * 10) / 10); return Math.max(0.5, x); };
    const aR = spR(b / 2, 45);
    const Lx = num(input.longueur), lx = num(input.largeur), L2x = num(input.longueur2), l2x = num(input.largeur2);
    const wb = /piscine|spa|bassin|pourtour/.test(g.forme) && Lx && lx && L2x && l2x ? Math.min((Lx - L2x) / 2, (lx - l2x) / 2) : 0;
    if (wb > 0 && wb <= 1.2) {
      // Pourtour étroit : une seule ligne de vis au milieu de la bande, sous longeron doublé, lames en porte-à-faux de part et d'autre
      const aC = spR(wb, 90), lc = (Lx + L2x) + (lx + l2x);
      nVis = Math.ceil(lc / aC) + 4; b = r2(wb); a = aC;
      hypotheses.push('Pourtour de ' + r2(wb) + ' m de large : une ligne centrale de vis sous longeron doublé tous les ' + aC + ' m sur ' + r2(lc) + ' m de développé, plus 4 vis d\'angle.');
    } else {
      nVis = Math.ceil(S / (a * b) + (P + Pint) / (2 * aR) + 1);
      hypotheses.push('Forme non rectangulaire : nombre de vis estimé à partir de la surface (une vis pour ' + r2(a * b) + ' m² sous les jonctions de cadres) et du périmètre (rives sur vis tous les ' + aR + ' m), à préciser sur plan.');
    }
  } else {
    nVis = Math.ceil(S / (a * b) + ((P + Pint) * (1 - deport / a)) / (a + b) + 1);
  }
  const Svis = /mural/i.test(String(input.adossee || '')) ? S / 2 : S; // avec muralière, la moitié de la charge va au bâtiment
  if (Svis / nVis > surfMaxVis) nVis = Math.ceil(Svis / surfMaxVis);
  let nVisSpa = 0;
  if (spa) { nVisSpa = R.spa_vis_supplement; alertes.push('Spa ou charge ponctuelle : ' + nVisSpa + ' vis supplémentaires avec bases de répartition ; au-delà de 1 500 kg, étude.'); }
  const nVisTotal = nVis + nVisSpa;
  hypotheses.push(nVisTotal + ' vis de fondation pour ' + S + ' m²' + (/mural/i.test(String(input.adossee || '')) ? ', la muralière reprenant la moitié de la charge : ' + r2(S / 2 / nVisTotal) + ' m² par vis' : ', soit une vis pour ' + r2(S / nVisTotal) + ' m²') + ' (plafond ' + r2(surfMaxVis) + ' m² par vis à ' + chargeM2 + ' kg/m² d\'exploitation plus ' + (num(R.charge_permanente) || 50) + ' kg/m² de poids propre, ' + capacite + ' kg par vis).');

  // --- Lignes ---
  const lignes = { t1: [], t2: [], t3: [] };
  const wItem = nom => W[String(nom).toLowerCase()];
  const ligneW = (t, nom, qte, designation) => {
    const it = wItem(nom);
    if (!it) { manquants.push('Weasyfix : ' + nom); return; }
    const pu = prixPalier(it, qte);
    if (pu === null) { manquants.push('Weasyfix : prix de ' + nom); return; }
    lignes[t].push({ designation: designation || (it.nom + ' — ' + it.description).slice(0, 90), qte, unite: 'pièce', pu, total: r2(qte * pu), ref: it.ref });
  };
  const ligneL = (t, cle, qte, unite, designation) => {
    const it = LPH[cle];
    if (!it) { manquants.push('LPH : ' + cle); return; }
    const pu = prixPalier(it, qte);
    if (pu === null) { manquants.push('LPH : prix de ' + cle); return; }
    lignes[t].push({ designation: designation || it.designation, qte: r2(qte), unite: unite || it.unite || '', pu, total: r2(qte * pu), cle });
  };
  const chutes = 1 + ((num(LPH.chutes_bois && LPH.chutes_bois.prix) ?? 8) / 100);

  // Grès cérame : forfaits WeasyCeram au m² (fondations + connecteurs, structure), vis comptées au m²
  if (gres) {
    if (S < (num(R.ceram_surface_min) || 5)) return etude('terrasse en dalles de moins de ' + (num(R.ceram_surface_min) || 5) + ' m² : étude');
    nVis = Math.ceil(S * (num(R.ceram_vis_par_m2) || 0.5));
    hypotheses.splice(-1, 1, nVis + nVisSpa + ' vis de fondation pour ' + S + ' m² de dalles (trame WeasyCeram, une vis pour ' + r2(1 / (num(R.ceram_vis_par_m2) || 0.5)) + ' m²).');
  }
  const nVisT = gres ? nVis + nVisSpa : nVisTotal;
  // Tiroir 1 : fondations posées
  if (gres) {
    const adj = { 'fs750-f': 'ceram_ajust_fs750', 'fs1250-f': 'ceram_ajust_fs1250', 'fs1120-fl': 'ceram_ajust_fs1120', 'fs1500-f': 'ceram_ajust_fs1500', 'fs1750-f': 'ceram_ajust_fs1750' }[visRef];
    const it = LPH.ceram_fondations_m2;
    if (it && num(it.prix) !== null) {
      const coef = num(LPH.ceram_coef && LPH.ceram_coef.prix) || 1.55;
      const pu = r2(num(it.prix) + (adj && num(R[adj]) ? num(R[adj]) * coef : 0));
      lignes.t1.push({ designation: 'Fondations WeasyCeram : vis ' + visRef + ', connecteurs wood basic 3D et fixations (' + nVisT + ' vis)', qte: S, unite: 'm²', pu, total: r2(S * pu), cle: 'ceram_fondations_m2' });
    } else manquants.push('LPH : ceram_fondations_m2');
    if (spa) ligneW('t1', 'base2 galva', 2, 'Bases de répartition sur 2 vis sous charge ponctuelle');
  } else {
  ligneW('t1', visRef, nVisTotal, 'Vis de fondation Weasyfix ' + visRef + ', acier galvanisé à chaud');
  const connecteur = gres ? null : ((unNiveau || R.poutre_defaut === '45x145') ? R.connecteur_solive : R.connecteur_poutre);
  if (connecteur) ligneW('t1', connecteur, nVisTotal, 'Connecteurs ' + connecteur + ' réglables, fixation sur tête de vis');
  if (spa) ligneW('t1', 'base2 galva', 2, 'Bases de répartition sur 2 vis sous charge ponctuelle');
  }
  ligneL('t1', 'mo_vis', nVisT, 'vis', 'Pose des vis : traçage, avant-trou de sondage, vissage, réglage de niveau');
  if (/roch|pierr|calcaire|schiste|dur/.test(sol)) ligneL('t1', 'mo_vis_rock', nVisT, 'vis');
  if (pente > R.pente_max_auto || /pi[eé]ton|[eé]troit|difficile/.test(String(input.acces || ''))) ligneL('t1', 'mo_vis_pente', nVisT, 'vis');
  ligneL('t1', 'plan_implantation', 1, 'forfait');
  // Test de compacité du sol sur site (visite de site), prix selon la distance aller-retour
  const tc = LPH.test_compacite, tcl = LPH.test_compacite_loin;
  if (/non|d[ée]j[àa]/i.test(String(input.test_compacite || ''))) hypotheses.push('Test de compacité du sol déjà réalisé sur ce site : non compté.');
  else if (tc && num(tc.prix) !== null) {
    let pu = prixPalier({ ...tc, conditionnement: 'pièce' }, kmAR);
    if (tcl && num(tcl.prix) !== null && num(tcl.q2) !== null && kmAR >= num(tcl.q2)) pu = num(tcl.prix);
    lignes.t1.push({ designation: 'Test de compacité du sol sur site (visite de site' + (kmAR ? ', ' + Math.round(kmAR) + ' km aller-retour' : '') + ')', qte: 1, unite: 'forfait', pu, total: pu, cle: 'test_compacite' });
  } else manquants.push('LPH : test_compacite');
  // Déplacement
  const dep = LPH.deplacement_forfait, depKm = LPH.deplacement_km, depIncl = num(LPH.deplacement_km_inclus && LPH.deplacement_km_inclus.prix) || 0;
  if (dep && num(dep.prix) !== null) {
    lignes.t1.push({ designation: 'Déplacement équipe et matériel (forfait' + (kmAR ? ', ' + Math.round(kmAR) + ' km aller-retour' : '') + ')', qte: 1, unite: 'forfait', pu: num(dep.prix), total: num(dep.prix), cle: 'deplacement_forfait' });
    if (depKm && num(depKm.prix) !== null && kmAR > depIncl) lignes.t1.push({ designation: 'Kilomètres au-delà du forfait (' + Math.round(kmAR - depIncl) + ' km AR)', qte: r2(kmAR - depIncl), unite: 'km', pu: num(depKm.prix), total: r2((kmAR - depIncl) * num(depKm.prix)), cle: 'deplacement_km' });
  } else manquants.push('LPH : deplacement_forfait');

  // Tiroir 2 : sous-structure posée
  if (gres) {
    const std = !/autre|40|30|80/.test(String(input.dalles || 'standard'));
    ligneL('t2', std ? 'ceram_structure_m2_std' : 'ceram_structure_m2_autres', S, 'm²', 'Structure WeasyCeram : profilés C62 Magnelis, jonctions, fixations, supports connect stone et stone clips' + (std ? ' (dalles 60/60, 60/120 ou 120/120)' : ' (autres formats de dalles)'));
    const moC = LPH.mo_ceram_m2, moV = LPH.mo_vis;
    if (moC && num(moC.prix) !== null) {
      const total = S * num(moC.prix), dejaVis = moV && num(moV.prix) !== null ? nVisT * num(moV.prix) : 0;
      const reste = Math.max(0, r2(total - dejaVis));
      if (reste > 0) lignes.t2.push({ designation: 'Pose et réglage de la structure (rendement WeasyCeram, déduction de la pose des vis déjà comptée)', qte: S, unite: 'm²', pu: r2(reste / S), total: reste, cle: 'mo_ceram_m2' });
    } else manquants.push('LPH : mo_ceram_m2');
    hypotheses.push('Terrasse en dalles sur un seul niveau, sens de pose unique sans décalage, dalles d\'au moins 30 cm (conditions WeasyCeram) ; habillage compté à part.');
  } else {
    // Quantités : sur un rectangle, pièces entières aux longueurs vendues ; sinon mètres linéaires + chutes
    const rect = g.forme === 'rectangle' && num(input.longueur) && num(input.largeur);
    const Lr = rect ? Math.max(num(input.longueur), num(input.largeur)) : 0, lr = rect ? Math.min(num(input.longueur), num(input.largeur)) : 0;
    const len45 = num(R.longueur_45x145) || 4.2, len70 = num(R.longueur_70x145) || 4;
    let mlSolives, mlPoutres = 0, piecesTxt = '';
    if (rect) {
      // Pièces entières aux longueurs vendues, chutes réemployées d'une ligne à l'autre (+8 % de perte)
      const nSol = Math.ceil(Lr / eS) + 1;
      const nJum = 0, mlRive = 0;
      const mlSol = unNiveau && calepinage && calepinage.cadres ? calepinage.cadres.n * calepinage.cadres.ml : nSol * lr + 2 * (Lr + lr);
      const p45 = Math.ceil(mlSol * 1.08 / len45);
      mlSolives = r2(p45 * len45);
      piecesTxt = p45 + ' pièces de 45 x 145 en ' + len45 + ' m';
      if (!unNiveau) { const nLines = Math.ceil((lr - 2 * deport) / b) + 1; const p70 = Math.ceil(nLines * Lr * 1.08 / len70); mlPoutres = r2(p70 * len70); piecesTxt += ', ' + p70 + ' pièces de ' + R.poutre_defaut + ' en ' + len70 + ' m'; }
      hypotheses.push('Bois compté en pièces entières aux longueurs vendues, chutes réemployées : ' + piecesTxt + '.');
    } else {
      mlSolives = r2((S / eS + P + (unNiveau ? S / b : 0)) * chutes); // un niveau : longerons doublés aux jonctions de cadres (S / b ml en plus)
      if (!unNiveau) mlPoutres = r2((S / b + P * 0.5) * chutes);
      hypotheses.push('Bois compté en mètres linéaires avec ' + Math.round((chutes - 1) * 100) + ' % de chutes (forme non rectangulaire).');
    }
    if (unNiveau) {
      const nCadres = calepinage && calepinage.cadres ? calepinage.cadres.n : Math.ceil(S / ((num(R.cadre_longueur_max) || 5) * b));
      ligneL('t2', 'bois_45x145', mlSolives, 'ml', 'Cadres 45 x 145 : longerons et solives tous les ' + Math.round(eS * 100) + ' cm (fourniture, ' + (calepinage && calepinage.cadres ? '' : 'environ ') + nCadres + ' cadres)');
      if (LPH.transport_cadres && num(LPH.transport_cadres.prix) !== null) lignes.t2.push({ designation: 'Fabrication des cadres en atelier et transport sur site', qte: 1, unite: 'forfait', pu: num(LPH.transport_cadres.prix), total: num(LPH.transport_cadres.prix), cle: 'transport_cadres' });
      else manquants.push('LPH : transport_cadres');
    } else {
      ligneL('t2', R.poutre_defaut === '70x195' ? 'bois_70x195' : R.poutre_defaut === '45x145' ? 'bois_45x145' : 'bois_70x145', mlPoutres, 'ml', 'Poutres ' + R.poutre_defaut + ' classe 4 sur connecteurs, lignes tous les ' + b + ' m (fourniture)');
      ligneL('t2', 'bois_45x145', mlSolives, 'ml', 'Solives 45 x 145 classe 4 tous les ' + Math.round(eS * 100) + ' cm, ceinture de rive comprise (fourniture)');
    }
    if (/mural/i.test(String(input.adossee || ''))) {
      const Lm = rect ? Lr : Math.sqrt(S) * 1.2;
      ligneL('t2', 'muraliere_ml', Lm, 'ml', 'Muralière 45 x 145 classe 4 fixée sur le bâtiment existant (chevilles ou tirefonds)');
      alertes.push('Fixation sur le bâtiment existant : la muralière reprend la moitié de la charge, l\'état du support (lisse basse, bardage, ossature) est vérifié à la visite.');
    }
    var mlStructure = mlSolives + mlPoutres;
    if (LPH.bande_etancheite && num(LPH.bande_etancheite.prix) !== null) ligneL('t2', 'bande_etancheite', r2(mlStructure + (/mural/i.test(String(input.adossee || '')) ? (rect ? Lr : 0) : 0)), 'ml', 'Bande d\'étanchéité EPDM sur la structure');
    ligneL('t2', 'mo_structure_m2', S, 'm²', unNiveau ? 'Assemblage en atelier, pose et réglage de niveau des cadres' : 'Pose et réglage de niveau de la sous-structure');
  }

  // Tiroir 3 : platelage et habillage
  let m2Lames = S;
  if (!gres && g.forme === 'rectangle' && num(input.longueur) && num(input.largeur)) {
    const Lr = Math.max(num(input.longueur), num(input.largeur)), lr = Math.min(num(input.longueur), num(input.largeur));
    const lenLame = num(R.longueur_lame) || 4.5, largLame = num(R.largeur_lame) || 0.15;
    const rows = Math.ceil(lr / largLame), pieces = Math.ceil(rows * Lr * 1.08 / lenLame);
    m2Lames = r2(pieces * lenLame * (largLame - 0.005));
    hypotheses.push('Lames : ' + rows + ' rangées, ' + pieces + ' lames de ' + lenLame + ' m (chutes réemployées), soit ' + m2Lames + ' m² achetés pour ' + S + ' m² posés.');
  } else if (!gres) m2Lames = r2(S * chutes);
  if (gres) ligneL('t3', 'gres_cerame', S, 'm²');
  else ligneL('t3', composite ? 'lames_composite' : exotique ? 'lames_exotique' : /douglas/.test(platelage) ? 'lames_douglas' : 'lames_pin', m2Lames, 'm²');
  if (/non|sans/i.test(String(input.habillage || ''))) { hypotheses.push('Habillage périphérique non compris.'); }
  else {
  const hab = hauteur < 0.15 ? 'habillage_rive' : hauteur <= 0.4 ? 'habillage_lames' : 'habillage_claire_voie';
  const Phab = input.adossee && input.adossee !== 'non' && g.forme === 'rectangle' && num(input.longueur) && num(input.largeur) ? r2(P - Math.max(num(input.longueur), num(input.largeur))) : P;
  ligneL('t3', hab, Phab + Pint, 'ml', 'Habillage périphérique (' + Math.round(hauteur * 100) + ' cm de hauteur, ' + r2(Phab + Pint) + ' ml' + (Phab < P ? ', côté bâtiment exclu' : '') + ')');
  }
  if (!gres && LPH.visserie_forfait && num(LPH.visserie_forfait.prix) !== null) ligneL('t3', 'visserie_forfait', 1, 'forfait', 'Visserie inox des lames');

  // Jours de chantier estimés par tiroir (rendements de l'onglet LPH) → hébergement et restauration si chantier éloigné
  const pv = k => num(LPH[k] && LPH[k].prix);
  const equipe = pv('equipe') || 2;
  const j1 = pv('vis_par_jour') ? nVisTotal / pv('vis_par_jour') : 0;
  const j2 = gres ? Math.max(0, S / (pv('ceram_m2_par_jour') || 30) - j1) : (pv('structure_m2_par_jour') ? S / pv('structure_m2_par_jour') : 0);
  const rLames = exotique ? (pv('lames_exotique_m2_par_jour') || pv('lames_m2_par_jour')) : pv('lames_m2_par_jour');
  const j3 = (rLames ? S / rLames : 0) + (pv('habillage_ml_par_jour') ? (P + Pint) / pv('habillage_ml_par_jour') : 0);
  const jours = [Math.ceil(j1), Math.ceil(j1 + j2), Math.ceil(j1 + j2 + j3)];
  hypotheses.push('Durée estimée : ' + jours[0] + ' jour(s) pour les fondations, ' + jours[1] + ' avec la structure, ' + jours[2] + ' pour la terrasse complète, à ' + equipe + ' personnes.');
  if (pv('hebergement_jour_pers') && pv('hebergement_si_trajet_min') && trajetMin >= pv('hebergement_si_trajet_min')) {
    const heb = pv('hebergement_jour_pers');
    const add = (t, d, lbl) => { if (d > 0) lignes[t].push({ designation: 'Hébergement et restauration de l\'équipe (' + d + ' jour(s) × ' + equipe + ' personnes' + lbl + ')', qte: d * equipe, unite: 'nuit', pu: heb, total: r2(d * equipe * heb), cle: 'hebergement' }); };
    add('t1', jours[0], ''); add('t2', jours[1] - jours[0], ' en plus'); add('t3', jours[2] - jours[1], ' en plus');
    alertes.push('Chantier à ' + trajetMin + ' min de route : hébergement et restauration comptés.');
  }
  const somme = arr => r2(arr.reduce((s, x) => s + x.total, 0));
  const t1 = somme(lignes.t1), t2 = somme(lignes.t2), t3 = somme(lignes.t3);
  const basse = num(LPH.fourchette_basse && LPH.fourchette_basse.prix) ?? 0.9, haute = num(LPH.fourchette_haute && LPH.fourchette_haute.prix) ?? 1.15;
  const tva = (num(LPH.tva && LPH.tva.prix) ?? 20) / 100;
  const tiroir = (nom, sousTitre, lignesT, cumulHT) => ({ nom, sousTitre, lignes: lignesT, ht: r2(cumulHT), ht_bas: r2(cumulHT * basse), ht_haut: r2(cumulHT * haute), ttc_bas: r2(cumulHT * basse * (1 + tva)), ttc_haut: r2(cumulHT * haute * (1 + tva)) });
  return {
    ok: true, etude: false,
    geometrie: g, nVis: nVisT, visRef, calepinage, structure: gres ? 'gres' : unNiveau ? 'un_niveau' : 'deux_niveaux', usage,
    tiroirs: [
      tiroir('Fondations posées', 'Vis dimensionnées, connecteurs, pose, plan d\'implantation et rapport de pose. Vous montez votre structure et vos lames.', lignes.t1, t1),
      tiroir('Fondations et sous-structure', 'Le tiroir 1 plus la structure bois posée et réglée, prête à recevoir les lames.', lignes.t2, t1 + t2),
      tiroir('Terrasse complète', 'Le tiroir 2 plus le platelage, l\'habillage périphérique et les finitions. Vous n\'avez rien à faire.', lignes.t3, t1 + t2 + t3),
    ],
    tva: tva * 100, basse, haute, alertes, hypotheses, manquants: [...new Set(manquants)],
    validite_jours: num(LPH.validite_jours && LPH.validite_jours.prix) ?? 30,
  };
}

// ---------- Pré-chiffrage Vistech (chalet, maison ossature bois, extension) ----------
function chiffrerVistech(input, tarifs) {
  const R = { vistech_entraxe_pieux: 2.2, vistech_largeur_refend: 5.5, vistech_taux_ext_bas: 0.1, vistech_taux_ext_haut: 0.5, vistech_taux_ext_bas_alt: 0.5, vistech_taux_ext_haut_alt: 1, vistech_max_pieux_auto: 40, vistech_hors_sol_max: 0.3, vistech_implantation_defaut: 0, trajet_max_auto: 180, ...(tarifs.regles || {}) };
  const V = tarifs.vistech || {};
  const LPH = tarifs.lph || {};
  const alertes = [], hypotheses = [], manquants = [];
  const etude = raison => ({ ok: false, etude: true, raison, alertes, hypotheses });
  const L = num(input.longueur), l = num(input.largeur);
  let S = num(input.surface);
  if (L && l) S = L * l;
  if (!S) return etude('emprise inconnue : demander les dimensions au sol');
  const Ls = L || Math.sqrt(S) * 1.2, ls = l || S / Ls;
  const altitude = num(input.altitude) || 0;
  const sol = String(input.sol || 'terre').toLowerCase();
  const horsSol = num(input.hors_sol) === null ? 0.3 : num(input.hors_sol);
  const trajetMin = num(input.trajet_min) || 0;
  const dept = String(input.code_postal || '').slice(0, 2);
  const jambes = horsSol > R.vistech_hors_sol_max;
  if (jambes && !(V.jambe_force && num(V.jambe_force.prix) !== null)) return etude('plancher à plus de ' + Math.round(R.vistech_hors_sol_max * 100) + ' cm hors sol : contreventement par jambes de force à tarifer (étude)');
  if (/remblai|non stabilis|tourbe/.test(sol)) return etude('sol remblayé, tourbeux ou non stabilisé : étude géotechnique');
  if (trajetMin > R.trajet_max_auto) return etude('temps de route supérieur à ' + R.trajet_max_auto + ' min : devis spécifique');
  if (/public|erp|collectiv/.test(String(input.usage || ''))) return etude('usage public ou ERP : étude');

  // Lignes porteuses : périmètre + refends selon la largeur
  const e = R.vistech_entraxe_pieux;
  const refends = ls > 2 * R.vistech_largeur_refend ? 2 : ls > R.vistech_largeur_refend ? 1 : 0;
  const nPeri = Math.ceil(2 * (Ls + ls) / e);
  const nRef = refends * (Math.ceil(Ls / e) + 1);
  const nPieux = nPeri + nRef;
  if (nPieux > R.vistech_max_pieux_auto) return etude(nPieux + ' pieux estimés, au-delà de ' + R.vistech_max_pieux_auto + ' : étude');
  hypotheses.push('Emprise ' + r2(Ls) + ' x ' + r2(ls) + ' m (' + r2(S) + ' m²) : pieux tous les ' + e + ' m sous les murs porteurs périphériques' + (refends ? ' et ' + refends + ' ligne(s) porteuse(s) intérieure(s)' : '') + ', soit ' + nPieux + ' pieux. La descente de charges du constructeur fixe le nombre et le diamètre définitifs.');
  const alt = altitude >= 800;

  if (/argil|humide|meuble|sabl/.test(sol)) { tBas = Math.min(1, tBas + 0.2); tHaut = Math.min(1.5, tHaut + 0.3); alertes.push('Sol ' + sol + ' : ancrage probablement plus profond, hypothèse haute relevée.'); }

  const htBas = r2(lignes.reduce((a, x) => a + x.total, 0)), htHaut = r2(lignes.reduce((a, x) => a + x.total_haut, 0));

    validite_jours: num(LPH.validite_jours && LPH.validite_jours.prix) ?? 30,
  };
}

// Aiguillage par famille de projet
function chiffrerProjet(input, tarifs) {
  const fam = String(input.famille || '').toLowerCase();
  const txt = [input.projet, input.systeme].filter(Boolean).join(' ').toLowerCase();
  if (fam === 'vistech' || (!fam && /vistech|chalet|maison|ossature|extension|agrandissement|v[ée]randa|reprise/.test(txt))) return chiffrerVistech(input, tarifs);
  if (fam === 'vrd' || (!fam && /vrd|terrassement|r[ée]seau|assainissement/.test(txt))) return { ok: false, etude: true, raison: 'VRD et terrassement : pas encore de chiffrage automatique, étude', alertes: [], hypotheses: [] };
  if (fam === 'leger' || (!fam && /abri|carport|pergola|studio|pool ?house|cabane|cl[oô]ture|ponton|passerelle|solaire/.test(txt))) return { ok: false, etude: true, raison: 'structure légère Weasyfix (abri, carport, pergola…) : pas encore de chiffrage automatique, étude', alertes: [], hypotheses: [] };
  return chiffrer(input, tarifs);
}

// Lecture des valeurs brutes des trois onglets (tableaux de lignes) → structure du moteur
function tarifsParse(sheets) {
  const idx = (head, name) => head.findIndex(h => String(h || '').toLowerCase().startsWith(name.toLowerCase()));
  const out = { weasyfix: {}, lph: {}, regles: {}, manquants: [] };
  const w = sheets.Weasyfix || [];
  if (w.length) {
    const h = w[0];
    const c = { ref: idx(h, 'Réf'), nom: idx(h, 'Nom'), desc: idx(h, 'Description'), cond: idx(h, 'Conditionnement'), pieces: idx(h, 'Pièces par'), prix: idx(h, 'Prix public'), q2: idx(h, 'Qté palier 2'), p2: idx(h, 'Prix palier 2'), q3: idx(h, 'Qté palier 3'), p3: idx(h, 'Prix palier 3'), used: idx(h, 'Utilisé') };
    for (const r of w.slice(1)) {
      const nom = String(r[c.nom] || '').trim(); if (!nom) continue;
      const it = { ref: r[c.ref], nom, description: String(r[c.desc] || '').slice(0, 120), conditionnement: r[c.cond], pieces: r[c.pieces], prix: r[c.prix], q2: r[c.q2], p2: r[c.p2], q3: r[c.q3], p3: r[c.p3], utilise: /oui/i.test(String(r[c.used] || '')) };
      out.weasyfix[nom.toLowerCase()] = it;
      if (it.utilise && num(it.prix) === null) out.manquants.push('Weasyfix : prix de ' + nom);
    }
  }
  const l = sheets.LPH || [];
  if (l.length) {
    const h = l[0];
    const c = { cle: idx(h, 'Clé'), des: idx(h, 'Désignation'), unite: idx(h, 'Unité'), prix: idx(h, 'Prix public'), q2: idx(h, 'Qté palier 2'), p2: idx(h, 'Prix palier 2'), q3: idx(h, 'Qté palier 3'), p3: idx(h, 'Prix palier 3') };
    for (const r of l.slice(1)) {
      const cle = String(r[c.cle] || '').trim(); if (!cle) continue;
      out.lph[cle] = { designation: r[c.des], unite: r[c.unite], prix: r[c.prix], q2: r[c.q2], p2: r[c.p2], q3: r[c.q3], p3: r[c.p3] };
      if (num(r[c.prix]) === null) out.manquants.push('LPH : ' + cle);
    }
  }
  const vt = sheets.Vistech || [];
  out.vistech = {}; out.reserves_vistech = [];
  if (vt.length) {
    const h = vt[0];
    const c = { cle: idx(h, 'Clé'), art: idx(h, 'Article'), unite: idx(h, 'Unité'), prix: idx(h, 'Prix public'), q2: idx(h, 'Qté palier 2'), p2: idx(h, 'Prix palier 2'), q3: idx(h, 'Qté palier 3'), p3: idx(h, 'Prix palier 3') };
    for (const r of vt.slice(1)) {
      const cle = String(r[c.cle] || '').trim(); if (!cle) continue;
      if (/^reserve_/.test(cle)) { if (r[c.art]) out.reserves_vistech.push(String(r[c.art])); continue; }
      if (cle === 'reserves') continue;
      out.vistech[cle] = { designation: r[c.art], unite: r[c.unite], prix: r[c.prix], q2: r[c.q2], p2: r[c.p2], q3: r[c.q3], p3: r[c.p3] };
      if (num(r[c.prix]) === null && !/sur devis/i.test(String(r[c.unite] || ''))) out.manquants.push('Vistech : ' + cle);
    }
  }
  const rg = sheets['Règles'] || sheets.Regles || [];
  if (rg.length) {
    const h = rg[0];
    const c = { cle: idx(h, 'Clé'), val: idx(h, 'Valeur') };
    for (const r of rg.slice(1)) {
      const cle = String(r[c.cle] || '').trim(); if (!cle) continue;
      const v = r[c.val]; const n = num(v);
      out.regles[cle] = (n !== null && /^[\d.,\s-]+$/.test(String(v))) ? n : String(v || '').trim();
    }
  }
  return out;
}


// ---------- Grille tarifaire : Google Sheets dans le Drive, lue par le moteur ----------
async function driveImportSheet(env, name, xlsxBytes, parentId, existingId) {
  const boundary = 'lphs' + Date.now().toString(36);
  const meta = JSON.stringify(existingId ? { name } : { name, mimeType: 'application/vnd.google-apps.spreadsheet', parents: [parentId] });
  const head = new TextEncoder().encode('--' + boundary + '\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n' + meta + '\r\n--' + boundary + '\r\nContent-Type: application/vnd.openxmlformats-officedocument.spreadsheetml.sheet\r\n\r\n');
  const tail = new TextEncoder().encode('\r\n--' + boundary + '--');
  const body = new Uint8Array(head.length + xlsxBytes.length + tail.length);
  body.set(head, 0); body.set(xlsxBytes, head.length); body.set(tail, head.length + xlsxBytes.length);
  const tok = await googleToken(env);
  const url = existingId ? 'https://www.googleapis.com/upload/drive/v3/files/' + existingId + '?uploadType=multipart&fields=id,name,webViewLink' : 'https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id,name,webViewLink';
  const res = await fetch(url, { method: existingId ? 'PATCH' : 'POST', headers: { Authorization: 'Bearer ' + tok, 'Content-Type': 'multipart/related; boundary=' + boundary }, body });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error('Drive import ' + res.status + ' ' + JSON.stringify(data).slice(0, 200));
  return data;
}

async function tarifsSheetInfo(env) { return env.CHAT_LOG ? env.CHAT_LOG.get('tarifs:sheet', 'json') : null; }

// Lecture des trois onglets ; cache 1 h dans KV (force = relecture immédiate)
async function tarifsLoad(env, force) {
  if (!env.CHAT_LOG) return null;
  if (!force) { const c = await env.CHAT_LOG.get('tarifs:data', 'json'); if (c) return c; }
  const info = await tarifsSheetInfo(env);
  if (!info || !info.id) return null;
  const ranges = ['Weasyfix!A1:P400', 'LPH!A1:I200', "Règles!A1:E200", 'Vistech!A1:I100'].map(r => 'ranges=' + encodeURIComponent(r)).join('&');
  const d = await gapi(env, 'https://sheets.googleapis.com/v4/spreadsheets/' + info.id + '/values:batchGet?' + ranges + '&valueRenderOption=UNFORMATTED_VALUE');
  const sheets = {};
  for (const vr of d.valueRanges || []) { const name = String(vr.range || '').split('!')[0].replace(/^'|'$/g, ''); sheets[name] = vr.values || []; }
  const T = tarifsParse(sheets);
  T.loaded = new Date().toISOString();
  T.sheet = info;
  await env.CHAT_LOG.put('tarifs:data', JSON.stringify(T), { expirationTtl: 3600 });
  return T;
}

// Altitude d'un point (service d'altimétrie IGN, gratuit)
async function altitudeOf(lat, lon) {
  try {
    const r = await fetch('https://data.geopf.fr/altimetrie/1.0/calcul/alti/rest/elevation.json?lon=' + lon + '&lat=' + lat + '&zonly=true', { headers: { 'User-Agent': 'les-pieux-horizon-chatbot' } });
    const j = await r.json();
    const z = j && j.elevations && j.elevations[0];
    const v = typeof z === 'object' ? z.z : z;
    return (typeof v === 'number' && v > -1000) ? Math.round(v) : null;
  } catch (e) { return null; }
}

// Chiffrage estimatif automatique depuis un lead du chatbot ou du formulaire : uniquement terrasse de particulier avec email, dimensions et hauteur
async function chiffrageAuto(env, lead, origin) {
  const l = lead || {};
  if (String(l.profil || 'particulier').toLowerCase() !== 'particulier') return { skipped: 'profil' };
  const input = leadToInput(l);
  if (input.famille !== 'terrasse') return { skipped: 'famille' };
  if (!input.email || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(input.email)) return { skipped: 'email' };
  if (!(num(input.longueur) && num(input.largeur)) && !num(input.surface)) return { skipped: 'dimensions' };
  if (num(input.hauteur) === null) return { skipped: 'hauteur' };
  let T = null; try { T = await tarifsLoad(env, false); } catch (e) { T = null; }
  if (!T) return { skipped: 'grille' };
  try {
    await homeGeo(env);
    const geo = await geocode(input.code_postal, input.ville);
    if (geo) {
      const alt = await altitudeOf(geo.lat, geo.lon); if (alt !== null) input.altitude = String(alt);
      const route = await routeMinutes(env, geo, input.code_postal);
      input.km_ar = String(Math.round((route.km || 0) * 2)); input.trajet_min = String(route.min);
    }
  } catch (e) { console.error('chiffrage auto geo', String(e)); }
  const r = chiffrerProjet(input, T);
  if (!r.ok) return { skipped: 'etude', raison: r.raison };
  if (r.manquants && r.manquants.length) return { skipped: 'manquants', raison: r.manquants.join(', ') };
  const sent = await chiffrageSend(env, { origin: origin || 'https://chatbot-pieux-horizon.a-ouzeau.workers.dev' }, input, r);
  return { remember: true, sent: true, lien: sent.lien, nVis: r.nVis, ht: r.tiroirs.map(t => t.ht), at: new Date().toISOString() };
}

  const vl = { id, civilite: input.civilite || '', prenom: input.prenom || '', nom: input.nom || '', email: input.email || '', telephone: input.telephone || '', code_postal: input.code_postal || '', ville: input.ville || '', type_projet: input.projet || 'Terrasse', surface: r.geometrie.S + ' m²', created: new Date().toISOString(), chiffrage: true };

'' : input.platelage, [input.code_postal, input.ville].filter(Boolean).join(' ') ? 'à ' + [input.code_postal, input.ville].filter(Boolean).join(' ') : ''].filter(Boolean).join(', ');
  const reserves = (r.reserves && r.reserves.length ? r.reserves : ['Prix HT, TVA en vigueur en sus ; chiffrage estimatif établi sur vos indications, devis définitif après visite terrain.', 'Sol conforme à l\'usage : en cas de refus (rocher, blocs, remblais), adaptation ou plus-value sur devis.', 'Repérage des réseaux enterrés à la charge du client.', 'Non compris : terrassement, démolition, évacuation, remise en état des espaces verts.']);
  const vars = { salutation: salutationFor(input.civilite, input.nom), nom: input.nom || '', prenom: input.prenom || '', projet, tiroirs: chiffrageTexte(r), hypotheses: r.hypotheses.concat(r.alertes).map(x => '- ' + x).join('\n'), reserves: reserves.map(x => '- ' + x).join('\n'), lien_visite: lien, validite: String(r.validite_jours), tva: String(r.tva) };
  const text = fillTemplate(t.chiffrage_body, vars);
  const htmlBody = brandedHtml(text).replace(escHtml(chiffrageTexte(r)).replace(/\n/g, '<br>'), chiffrageHtml(r));
  await gmailSend(env, { to: input.email, subject: fillTemplate(t.chiffrage_subject, vars), text, html: htmlBody, cc: env.CHIFFRAGE_CC || 'contact@chalets-horizon.com' });
  const rec = { id, at: new Date().toISOString(), client: vl, input, resume: r.tiroirs.map(x => ({ nom: x.nom, ttc_bas: x.ttc_bas, ttc_haut: x.ttc_haut, ht: x.ht })), nVis: r.nVis, visRef: r.visRef, lien };
  await env.CHAT_LOG.put('chiffrage:' + id, JSON.stringify(rec), { expirationTtl: LOG_TTL_DAYS * 86400 });
  // HubSpot
  let hubspot = false;
  try {
    if (env.HUBSPOT_TOKEN) {
      const filterGroups = [];
      if (input.email) filterGroups.push({ filters: [{ propertyName: 'email', operator: 'EQ', value: input.email }] });
      if (input.telephone) filterGroups.push({ filters: [{ propertyName: 'phone', operator: 'EQ', value: input.telephone }] });
      const found = await hs('/crm/v3/objects/contacts/search', 'POST', { filterGroups, properties: ['email'], limit: 1 }, env);
      const contact = (found.results || [])[0];
      if (contact) {
        const txt = 'CHIFFRAGE ESTIMATIF envoyé le ' + new Date().toLocaleDateString('fr-FR', { timeZone: 'Europe/Paris' }) + ' : ' + r.tiroirs.map((x, i) => (i + 1) + '.         const deals = await hs('/crm/v3/objects/deals/search', 'POST', { filterGroups: [{ filters: [{ propertyName: 'associations.contact', operator: 'EQ', value: contact.id }] }], properties: ['dealstage', 'description'], sorts: [{ propertyName: 'hs_lastmodifieddate', direction: 'DESCENDING' }], limit: 5 }, env).catch(() => ({ results: [] }));
        const deal = (deals.results || []).find(d => !['closedwon', 'closedlost'].includes(d.properties.dealstage));
        const assoc = [{ to: { id: contact.id }, types: [{ associationCategory: 'HUBSPOT_DEFINED', associationTypeId: 202 }] }];
        if (deal) {

  return { lien, hubspot };

function todayFr() {
  return new Date().toLocaleDateString('fr-FR', { timeZone: 'Europe/Paris', weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });
}

async function sha256Hex(s) {
  const h = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s));
  return [...new Uint8Array(h)].map(b => b.toString(16).padStart(2, '0')).join('');
}

// Identifiant stable d'une conversation : le site renvoie tout l'historique à chaque tour,
// on dérive donc l'identifiant du premier message du visiteur, de son adresse IP et du jour.
async function conversationId(request, messages, body) {
  const first = messages.find(m => m.role === 'user');
  const ip = request.headers.get('CF-Connecting-IP') || '';
  const ua = request.headers.get('User-Agent') || '';
  const day = new Date().toISOString().slice(0, 10);
  if (body && typeof body.convId === 'string' && /^[a-z0-9-]{6,40}$/i.test(body.convId)) return body.convId;
  return (await sha256Hex(ip + '|' + ua + '|' + day + '|' + (first ? String(first.content || '').slice(0, 200) : ''))).slice(0, 16);
}

function stripLead(t) { return String(t || '').replace(/\[LEAD\][\s\S]*$/, '').trim(); }

async function logConversation(convId, transcript, lead, request, env) {
  if (!env.CHAT_LOG) return;
  try {
    const prev = (await env.CHAT_LOG.get('conv:' + convId, 'json')) || {};
    const now = new Date().toISOString();
    const page = request.headers.get('Referer') || prev.page || '';
    const messages = transcript.map(m => ({ role: m.role, content: stripLead(m.content) }));
    const rec = { id: convId, kind: 'chat', started: prev.started || now, updated: now, page, messages, lead: lead || prev.lead || null };
    const first = messages.find(m => m.role === 'user');
    const l = rec.lead || {};
    const metadata = {
      updated: now, lead: !!rec.lead,
      who: [l.prenom, l.nom].filter(Boolean).join(' ').slice(0, 60),
      projet: String(l.type_projet || '').slice(0, 40), cp: String(l.code_postal || '').slice(0, 10),
      preview: (first ? first.content : '').slice(0, 80),
      turns: messages.filter(m => m.role === 'user').length,
    };
    await env.CHAT_LOG.put('conv:' + convId, JSON.stringify(rec), { expirationTtl: LOG_TTL_DAYS * 86400, metadata });
  } catch (e) { console.error('Erreur logConversation', String(e && e.message ? e.message : e)); }
}

    const id = String(q.id);
    const prev = (await env.CHAT_LOG.get('qual:' + id, 'json')) || {};
    const now = new Date().toISOString();
    const rec = {
      id, kind: 'qualif', started: prev.started || now, updated: now,

      tags: Array.isArray(q.tags) ? q.tags.map(String).slice(0, 10) : [],
      changes: (prev.changes || 0) + 1,
      lead: prev.lead || null,
    };
    const metadata = {
      updated: now, lead: !!rec.lead, who: rec.lead ? [rec.lead.prenom, rec.lead.nom].filter(Boolean).join(' ').slice(0, 60) : '',
      projet: rec.structure.slice(0, 40), cp: rec.lead ? String(rec.lead.code_postal || '') : '',

      verdict: rec.verdict.slice(0, 60),
    };
    await env.CHAT_LOG.put('qual:' + id, JSON.stringify(rec), { expirationTtl: LOG_TTL_DAYS * 86400, metadata });

}

      '<form method="post" action="/admin/tarifs' + q + '" enctype="multipart/form-data"><div class="box"><b>' + (info ?   // Chiffrage estimatif (formulaire, aperçu, envoi)
  if (id === 'chiffrage') {

  // Agenda des visites : réglages, agendas Google, test des créneaux, visites à venir

      catch (e) { msg = '<div class="box"><b>Erreur agendas</b>' + escHtml(String(e && e.message ? e.message : e)) + '</div>'; }

    const cp = (url.searchParams.get('cp') || '').replace(/\D/g, '').slice(0, 5);
    if (action === 'test' && cp) {
      try {
        const r = await agendaSlots(env, { code_postal: cp, ville: url.searchParams.get('ville') || '' }, null, 'lien');
        msg = '<div class="box"><b>Test pour ' + escHtml(cp) + (r.commune ? ' ' + escHtml(r.commune) : '') + '</b>' + (r.ok
          ? 'Secteur ' + escHtml(r.secteur) + ', ' + r.trajet_min + ' min de route' + (r.trajet_km ? ' (' + r.trajet_km + ' km)' : '') + (r.trajet_source === 'maps' ? ' selon Google Maps' : ' estimées (clé GOOGLE_MAPS_KEY absente : estimation à vol d\'oiseau)') + ' depuis ' + escHtml(HOME_ADDRESS) + '. Créneaux proposés : ' + (r.creneaux.length ? r.creneaux.map(c => escHtml(c.libelle)).join(', ') : 'aucun') + '.'
          : 'Pas de créneau (' + escHtml(r.raison) + (r.trajet_min ? ', ' + r.trajet_min + ' min de route' + (r.trajet_km ? ', ' + r.trajet_km + ' km' : '') : '') + ') : ' + escHtml(r.consigne || '')) + '</div>';
      } catch (e) { msg = '<div class="box"><b>Erreur test</b>' + escHtml(String(e && e.message ? e.message : e)) + '</div>'; }
    }
    const cfg = await agendaConfig(env);
    const cal = await agendaCalendars(env);
    let upcoming = '';
    if (cal && cal.visites) {
      try {
        const list = await agendaUpcoming(env, 60);
        upcoming = list.length ? '<table><thead><tr><th>Date</th><th>Visite</th><th>Lieu</th><th>Secteur</th></tr></thead><tbody>' + list.map(v => '<tr><td><small>' + escHtml(libelleFr(new Date(v.start))) + '</small></td><td><a target="_blank" href="' + escHtml(v.link || '#') + '">' + escHtml(v.summary || '') + '</a></td><td>' + escHtml(v.location || '') + '</td><td>' + escHtml(v.secteur || '') + '</td></tr>').join('') + '</tbody></table>' : '<small>Aucune visite planifiée sur les 60 prochains jours.</small>';
      } catch (e) { upcoming = '<small>Lecture de l\'agenda impossible : ' + escHtml(String(e && e.message ? e.message : e)) + '</small>'; }
    }
    const inp = (name, val, w) => '<input name="' + name + '" value="' + escHtml(val) + '" style="width:' + (w || 80) + 'px;padding:6px 8px;border:1px solid var(--bd);border-radius:6px;font:14px Inter,system-ui,sans-serif">';
    const days = [[1, 'Lundi'], [2, 'Mardi'], [3, 'Mercredi'], [4, 'Jeudi'], [5, 'Vendredi'], [6, 'Samedi']].map(([n, l]) => '<label style="margin-right:14px"><input type="checkbox" name="jours" value="' + n + '"' + (cfg.jours.includes(n) ? ' checked' : '') + '> ' + l + '</label>').join('');
    return html('<a class="back" href="/admin' + q + '">← Tout le journal</a><h1>Agenda des visites</h1><p class="sub">Le chatbot propose et réserve les visites terrain dans l\'agenda Google « Visites ». Les agendas « Chantiers » et principal bloquent les créneaux ; « Matériel » sert au suivi des locations.</p>' + msg +
      '<div class="box"><b>1. Agendas Google</b>' + (cal && cal.visites ? 'En place depuis le ' + fmtDate(cal.created) + '. <a target="_blank" href="https://calendar.google.com/">Ouvrir Google Agenda</a> · <a href="/admin/agenda' + q + '&action=init">vérifier</a>' : (t ? '<a href="/admin/agenda' + q + '&action=init">Créer les agendas Visites, Chantiers et Matériel</a>' : '<small>Connectez Google d\'abord (<a href="/admin/google' + q + '">/admin/google</a>).</small>')) + '</div>' +
      '<form method="post" action="/admin/agenda' + q + '"><div class="box"><b>2. Règles de planification</b>' +
      '<div style="margin:8px 0">Jours de visite : ' + days + '</div>' +
      '<div style="margin:8px 0">Matin de ' + inp('matin_debut', cfg.matin[0]) + ' à ' + inp('matin_fin', cfg.matin[1]) + ' &nbsp; Après-midi de ' + inp('am_debut', cfg.apres_midi[0]) + ' à ' + inp('am_fin', cfg.apres_midi[1]) + '</div>' +
      '<div style="margin:8px 0">Durée d\'une visite ' + inp('duree', String(cfg.duree), 60) + ' min &nbsp; Trajet bloqué avant et après ' + inp('tampon', String(cfg.tampon), 60) + ' min </div>' +
      '<div style="margin:8px 0">Départ du domicile au plus tôt ' + inp('depart_min', cfg.depart_min) + ' &nbsp; Retour au plus tard ' + inp('retour_max', cfg.retour_max) + ' &nbsp; Garde-fou chatbot : pas de réservation au-delà de ' + inp('trajet', String(cfg.trajet_max_min), 60) + ' min de route (le lien envoyé avec le chiffrage n\'est pas limité)</div>' +
      '<div style="margin:8px 0">Premier créneau proposé à J+' + inp('delai', String(cfg.delai_min_jours), 50) + ' &nbsp; Horizon ' + inp('horizon', String(cfg.horizon_jours), 50) + ' jours &nbsp; Visites maximum par demi-journée ' + inp('max_dj', String(cfg.max_par_demi_journee), 50) + '</div>' +

Jours fériés exclus automatiquement.</small><br><br>' +
      '<button type="submit" style="background:var(--gr);color:#fff;border:0;padding:10px 18px;border-radius:8px;font:600 14px Inter,system-ui,sans-serif;cursor:pointer">Enregistrer</button></div></form>' +
      '<form method="get" action="/admin/agenda"><input type="hidden" name="token" value="' + escHtml(token) + '"><input type="hidden" name="action" value="test"><div class="box"><b>3. Tester les créneaux</b>Code postal ' + inp('cp', cp, 90) + ' Commune ' + inp('ville', url.searchParams.get('ville') || '', 160) + ' <button type="submit" style="background:var(--gr);color:#fff;border:0;padding:8px 14px;border-radius:8px;font:600 13px Inter,system-ui,sans-serif;cursor:pointer">Voir les créneaux</button></div></form>' +
      '<form method="post" action="/admin/agenda' + q + '"><input type="hidden" name="action" value="lien"><div class="box"><b>4. Lien de réservation à joindre au chiffrage estimatif</b>' +

  if (id === 'google') {

    if (action === 'drive-init' && t) {

      '<div class="box"><b>1. Connecter le compte</b>' + (t ? 'Déjà connecté. ' : '') + '<a href="/oauth/start' + q + '">Connecter ou reconnecter Google</a><br><small>Utilisez le compte contact@chalets-horizon.com.</small></div>' +
      '<div class="box"><b>2. Créer l\'arborescence Drive</b>' + (tree2 ? 'Créée le ' + fmtDate(tree2.created) + ' — <a target="_blank" href="https://drive.google.com/drive/folders/' + escHtml(tree2.root) + '">ouvrir</a>. ' : '') + (t ? '<a href="/admin/google' + q + '&action=drive-init">' + (tree2 && tree2.complete ? 'Vérifier / compléter' : tree2 ? 'Continuer la création' : 'Créer maintenant') + '</a>' : '<small>Connectez Google d\'abord.</small>') + '</div>' +

'Dernière relève : ' + fmtDate((await env.CHAT_LOG.get('inbox:last', 'json')).at) + '. ' : 'Jamais relevée. Modèles : <a href="/admin/emails' + q + '">/admin/emails</a>.</small>' : '<small>Connectez Google d\'abord.</small>') + '</div>' +
      '<div class="box"><b>5. Agenda des visites</b><a href="/admin/agenda' + q + '">Créer les agendas et régler les jours de visite</a></div>' +
      '<div class="box"><b>3. Tester l\'envoi d\'email</b>' + (t ?   // Export JSON de tout le journal
  if (id === 'export.json') {

    for (const prefix of ['conv:', 'qual:']) {

      } while (cursor);
    }
    return new Response(JSON.stringify(out, null, 2), { headers: { 'Content-Type': 'application/json; charset=utf-8', 'Content-Disposition': 'attachment; filename="conversations-chatbot.json"' } });

  // Détail d'une conversation
  if (id) {

'<div class="box"><b>Coordonnées laissées via le formulaire</b>' + [['Prénom', l.prenom], ['Nom', l.nom], ['Téléphone', l.telephone], ['Email', l.email], ['Code postal', l.code_postal]].filter(r => r[1]).map(r => '<div><small>' + r[0] + '</small> : ' + escHtml(r[1]) + '</div>').join('') + '</div>'
